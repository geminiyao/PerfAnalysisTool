72function cls:MeetScopeType(scopeType, sceneName)
73    if scopeType == ESubMgrScopeType.Always then
74        return true
75    elseif scopeType == ESubMgrScopeType.Default then
76        if sceneName == SceneName.Outside or sceneName == SceneName.Raid then
77            return true

80        return true

224---@private
225function cls:EndProcessZombie()
226    self._markZombieTime = 0
227
228    local zombieDataLength = 0
229    local allDatas = self.DataCtrl:GetAllDatas()
230    for entityId, entityData in pairs(allDatas) do
231        local isZombieState = entityData:IsZombieState()
232        if isZombieState then
233            local significanceTask = self.MapSignificanceMgr:GetEntitySignificanceTask(entityId, EMapSignificanceTaskType.MapObjInit)
234            if significanceTask then
235                isZombieState = false
236            end
237        end
238        if isZombieState then
239            zombieDataLength = zombieDataLength + 1
240            zombieDataIds[zombieDataLength] = entityId
241        end
242    end
243
244    for i = 1, zombieDataLength do
245        local entityId = zombieDataIds[i]
246        self:DestroyMapObject(entityId)
247    end
248end

466--region MapObject协议入口
467---@param entityAttr NetNs.proto_WorldEntityAttr
468---@param isFullUpdate boolean 是否为全量更新数据
469---@param isLocalData boolean 是否为本地Entity数据
470---@param viewLayer number 无极缩放层级
471---@return MapEntityBase
472function cls:ProcessEntityAttr(entityAttr, isFullUpdate, isLocalData, viewLayer)
473    -- mgr.garyLog:Info("[debug] ProcessEntityAttr : %s, %s, %s", entityAttr.entityId, entityAttr.entityType, isFullUpdate)
474    
475    -- 有配置的才创建
476    if entityAttr.entityType and not MapEntityCfg[entityAttr.entityType] then
477        return
478    end
479
480    -- 屏蔽旧版 WorldEntityNTF 资源田：WET_Resource 且无 resourceLandCompactAttr 组件
481    -- 新版数据走 GeoGridMgrAttr.resourceLandAttr → OutsideLocalResEntityHandler → ProcessEntityAttr（带 resourceLandCompactAttr 组件）
482    if entityAttr.entityType == CfgNs.WorldEntityType.WET_Resource then
483        local hasNewComp = entityAttr.compatibleComponentMap and entityAttr.compatibleComponentMap.resourceLandCompactAttr
484        if not hasNewComp then
485            return
486        end
487    end
488
489    ---@type MapEntityDataBase
490    local entityData = nil
491    if isFullUpdate and self.LocalEntityMgr and not isLocalData then -- 收到服务器数据，需要覆盖掉本地创建的Entity数据
492        self.LocalEntityMgr:OnServerEntityReceived(entityAttr)
493    end
494
495    if isFullUpdate then
496        entityData = self.DataCtrl:CreateEntityDataByEntityAttr(entityAttr, isLocalData, viewLayer)
497    else
498        entityData = self.DataCtrl:MergeDataFromEntityAttr(entityAttr, viewLayer)
499    end
500
501    if not entityData then
502        return
503    end
504
505    local isInView = self:IsMapEntityInView(entityData)
506    if isInView then
507        local entityId = entityData.entityId
508        local mapEntity = self.MapEntityCtrl:GetEntityById(entityId)
509        local infiniteZoomMgr = mgr.infiniteZoomMgr
510        local zoomEntityCtrl = infiniteZoomMgr.MapEntityCtrl
511        local significanceMgr = self.MapSignificanceMgr
512        local checkViewLayer = entityData:CheckViewLayer()
513        local canShowMapEntity = zoomEntityCtrl:CanShowMapEntityGo(entityData)
514        local battleNotShowEntity = mgr.battleMode:IsBattleFieldBlockOperate(entityId, OutsideConstDefine.BattleFieldOperateType.ModeVisible)
515        if canShowMapEntity and not battleNotShowEntity then
516            if checkViewLayer then
517                if isLocalData then
518                    if mapEntity == nil then
519                        significanceMgr:RequestTaskSync_MapEntityAdd(entityId)
520                    else
521                        -- mgr.garyLog:Info("LocalResourceEntity already exist : %s, gridIndex : %s", entityId, entityData.gridIndex)
522                    end
523                else
524                    if mapEntity == nil or mapEntity:IsZombieState() then
525                        significanceMgr:RequestTask_MapEntityAdd(entityId)
526                    else
527                        --#if ENABLE_PROFILER
528                        local sampler = self:GetSampler_RefreshMapEntity(entityData.entityType)
529                        ProfilerUtil.SamplerBegin(sampler)
530                        --#endif
531                        mapEntity:RefreshByEntityAttr(entityData)
532                        --#if ENABLE_PROFILER
533                        ProfilerUtil.SamplerEnd(sampler)
534                        --#endif
535                    end
536                end
537            else
538                -- mgr.log:Info("checkViewLayer invalid : %s, %s, %s, %s", entityData.entityId, entityData.entityType, entityData.viewLayer, mgr.infiniteZoomMgr:GetIconLayerLevel())
539            end
540        else
541            if mapEntity ~= nil or significanceMgr:GetEntitySignificanceTask(entityId, EMapSignificanceTaskType.MapEntityAdd) ~= nil then
542                significanceMgr:RequestTask_MapEntityRemove(entityId)
543            end
544        end
545
546        -- 图标创建暂时不依赖数据校验，若需要请将checkViewLayer放到CreateMapEntityIcon中
547        -- 仅玩家部队需要检查layer（处理部队摆脏字需求）
548        if InfiniteZoomUtil.IsLocalIconBigCityEntityType(entityData.entityType) and _G.ReqViewBigCity ~= false then
549            mgr.infiniteZoomMgr.MapGoEffectCtrl:RefreshBigCityServerEntityID(entityData)
550        else
551            local layerValid = checkViewLayer or entityData.entityType ~= CfgNs.WorldEntityType.WET_Army
552            local checkEntityType = self:NeedCheckAddIconEntity(entityData)
553            if layerValid and checkEntityType then
554                if isFullUpdate or not zoomEntityCtrl:GetMapEntityIconByEntityId(entityId) then
555                    significanceMgr:RequestTask_ZoomEntityAdd(entityId)
556                else -- 增量更新时，需立即（不能异步）触发图标层单位的刷新，因为此时经过 MergeDataFromEntityAttr 后的dirty标记是针对类型而非实例的
557                    zoomEntityCtrl:CreateMapEntityIcon(entityData)
558                end
559            end
560        end
561    else
562        self:DestroyMapEntity(entityData)
563    end

620    --#endif
621
622    --#if ENABLE_PROFILER
623    -- ProfilerUtil.SamplerBegin(sampler_DestroyMapObject)

681function cls:ScheduleInfiniteLayerEntityScan(allMapEntityData)
682    -- 只快照 ID；后续按 ID 取最新数据。连续跨层时新任务覆盖旧任务，最终层会全量校正。
683    local entityIds = self._infiniteLayerEntityIds or {}
684    for index = #entityIds, 1, -1 do
685        entityIds[index] = nil
686    end
687    for entityId in pairs(allMapEntityData) do
688        entityIds[#entityIds + 1] = entityId
689    end
690    self._infiniteLayerEntityIds = entityIds
691
692    self._infiniteLayerEntityScanTask = {
693        allMapEntityData = allMapEntityData,
694        entityIds = entityIds,
695        nextIndex = 1,
696        isDefaultLayer = mgr.infiniteZoomMgr:IsDefaultLayer(),
697    }
698end

700---@param task table
701---@param entityId number
702---@param entityData MapEntityDataBase
703function cls:ProcessInfiniteLayerEntity(task, entityId, entityData)
704    local entityCtrl = self.MapEntityCtrl
705    local significanceMgr = self.MapSignificanceMgr
706    local infiniteEntityCtrl = mgr.infiniteZoomMgr.MapEntityCtrl
707
708    -- 与 OnInfiniteLayerLevelChange 原有循环保持同一业务处理；仅改变执行批次。
709    if entityData.entityType == OutsideLocalEntityType.LocalStaticObstacleEntity then
710        if not task.isDefaultLayer then
711            local mapEntity = entityCtrl:GetEntityById(entityData.entityId)
712            if mapEntity ~= nil then
713                significanceMgr:RequestTask_MapEntityRemove(entityData.entityId)
714            end
715        end
716        return
717    end
718
719    local isInView = self:IsMapEntityInView(entityData)
720    if isInView then
721        local checkViewLayer = entityData:CheckViewLayer()
722        local canShowMapEntity = infiniteEntityCtrl:CanShowMapEntityGo(entityData)
723        local mapEntity = entityCtrl:GetEntityById(entityData.entityId)
724        if canShowMapEntity then
725            if checkViewLayer and (mapEntity == nil or mapEntity:IsZombieState()) then
726                entityData._needCallInfiniteZoomCallback = true
727                significanceMgr:RequestTask_MapEntityAdd(entityData.entityId)
728            end
729            if mapEntity ~= nil and mapEntity.OnInfiniteLayerLevelChange then
730                mapEntity:OnInfiniteLayerLevelChange()
731            end
732        elseif mapEntity ~= nil or
733            significanceMgr:GetEntitySignificanceTask(entityData.entityId, EMapSignificanceTaskType.MapEntityAdd) ~= nil then
734            significanceMgr:RequestTask_MapEntityRemove(entityData.entityId)
735        end
736    else
737        self:DestroyMapEntity(entityData)
738    end
739
740    if not task.isDefaultLayer then
741        if InfiniteZoomUtil.IsLocalIconBigCityEntityType(entityData.entityType) and _G.ReqViewBigCity ~= false then
742            mgr.infiniteZoomMgr.MapGoEffectCtrl:RefreshBigCityServerEntityID(entityData)
743        else
744            local checkViewLayer = entityData.entityType ~= CfgNs.WorldEntityType.WET_Army or entityData:CheckViewLayer()
745            local checkEntityType = self:NeedCheckAddIconEntity(entityData)
746            if checkViewLayer and checkEntityType then
747                significanceMgr:RequestTask_ZoomEntityAdd(entityId)
748            end
749        end
750    end
751end

753---@param flushAll boolean|nil
754function cls:ProcessInfiniteLayerEntityScanTask(flushAll)
755    local task = self._infiniteLayerEntityScanTask
756    if not task then
757        return
758    end
759
760    --#if ENABLE_PROFILER
761    ProfilerUtil.SamplerBegin(samplerInfiniteLayerEntityScanDeferred)
762    --#endif
763    local budget = flushAll and #task.entityIds or InfiniteLayerEntityScanBudgetPerFrame
764    local endIndex = math.min(#task.entityIds, task.nextIndex + budget - 1)
765    for index = task.nextIndex, endIndex do
766        local entityId = task.entityIds[index]
767        local entityData = task.allMapEntityData[entityId]
768        if entityData then
769            self:ProcessInfiniteLayerEntity(task, entityId, entityData)
770        end
771    end
772    task.nextIndex = endIndex + 1
773    if task.nextIndex > #task.entityIds then
774        self._infiniteLayerEntityScanTask = nil
775    end
776    --#if ENABLE_PROFILER
777    ProfilerUtil.SamplerEnd(samplerInfiniteLayerEntityScanDeferred)
778    --#endif
779end
