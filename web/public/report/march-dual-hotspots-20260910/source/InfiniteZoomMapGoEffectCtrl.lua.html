253function cls:OnUpdate()
254    if not mgr.scene:IsInScene(SceneName.Outside) or not mgr.mapCore?.MapViewCtrl then
255        return
256    end
257
258    --#if ENABLE_PROFILER
259    ProfilerUtil.SamplerBegin(sampler_CameraMoveDeferredRefresh)
260    --#endif
261    self:TryFlushCameraMoveRefresh()
262    --#if ENABLE_PROFILER
263    ProfilerUtil.SamplerEnd(sampler_CameraMoveDeferredRefresh)
264    --#endif
265
266    local state = mgr.infiniteZoomMgr:GetCurState()
267    local checkTouching = (state == InfiniteZoomState.OutsideIcon or state == InfiniteZoomState.Outside) and not self.touching
268    -- 分帧更新标记显示
269    --if self.refreshIndex < self.markCnt and checkTouching then --地表层移动地图不更新，等停止操作后再更新
270    if self.refreshIndex < self.markCnt then --只要标记出现在视野内就立刻更新
271        --#if ENABLE_PROFILER
272        ProfilerUtil.SamplerBegin(sampler_RefreshMarkVisble)
273        --#endif
274        local isDefaultOrIcon = state ~= InfiniteZoomState.PlanViewMode
275        local refreshEndIndex = self.markCnt
276        if CameraMoveRefreshOptimizationEnabled then
277            refreshEndIndex = math.min(self.markCnt, self.refreshIndex + CameraMoveMarkBudgetPerFrame)
278        end
279        for i = self.refreshIndex + 1, refreshEndIndex do
280            local v = self.markList[i]
281            if v then
282                local visible = (v.type == InfiniteZoomCfg.MarkType.Personal and self.personalMarkVisible) or (self.allianceMarkVisible) or (v.type == InfiniteZoomCfg.MarkType.DeclaredCity)
283                local isUnitInView = mgr.mapCore.MapViewCtrl:IsUnitInView(v.gridIndex)
284                if(SubMapUtil.IsOpenSubMap())then
285                    isUnitInView = isUnitInView or SubMapUtil.IsInSubMapRect(v.gridIndex)
286                end
287                visible = visible and isUnitInView and isDefaultOrIcon
288                self.refreshIndex = self.refreshIndex + 1
289                self:SetMarkDataVisible(v, visible) -- 每帧更新全部标记
290                --if self:SetMarkDataVisible(v, visible) then --涉及显示变化的标记一帧只更新一个
291                    -- mgr.mingdongLog:Error(changed)

293                --end
294            end
295        end
296        --#if ENABLE_PROFILER
297        ProfilerUtil.SamplerEnd(sampler_RefreshMarkVisble)
298        --#endif
299    end
300
301    -- 优化模式下，名城图标的创建/删除/刷新在这里按预算分帧执行。

1409function cls:ProcessBigCityIconRefreshTasks()
1410    if not BigCityIconDiffRefreshEnabled or
1411        not OutsideMapPerfDiagnostic.MapGoBigCityIconRefreshEnabled then
1412        return
1413    end
1414
1415    local tasks = self._bigCityIconRefreshTasks
1416    local taskIndex = self._bigCityIconRefreshTaskIndex or 1
1417    if not tasks or taskIndex > #tasks then
1418        return
1419    end
1420
1421    local processedCount = 0
1422    while taskIndex <= #tasks and processedCount < BigCityIconTaskBudgetPerFrame do
1423        local task = tasks[taskIndex]
1424        taskIndex = taskIndex + 1
1425        processedCount = processedCount + 1
1426
1427        local data = task.data
1428        local visible = data and self:IsBigCityIconVisible(data)
1429        local mapIcon = data and MapEntityCtrl:GetMapEntityIconByEntityId(data.entityId) or nil
1430        if task.taskType == BigCityIconTaskType.Add then
1431            if visible and not mapIcon then
1432                MapEntityCtrl:CreateMapEntityIcon(data)
1433            end
1434        elseif task.taskType == BigCityIconTaskType.Remove then
1435            if not visible and mapIcon then
1436                MapEntityCtrl:RemoveMapEntityIcon(data.entityId)
1437            end
1438        elseif task.taskType == BigCityIconTaskType.Refresh then
1439            if visible then
1440                if mapIcon then
1441                    self:RefreshExistingBigCityIcon(mapIcon, data)
1442                else
1443                    MapEntityCtrl:CreateMapEntityIcon(data)
1444                end
1445            elseif mapIcon then
1446                MapEntityCtrl:RemoveMapEntityIcon(data.entityId)
1447            end
1448        end
1449    end
1450
1451    self._bigCityIconRefreshTaskIndex = taskIndex
1452    if taskIndex > #tasks then
1453        self:ClearBigCityIconRefreshTasks()
1454    end
1455end

1813function cls:ShowRopeLine(mapEntityData,flag)

1815    local relation = mapEntityData:GetZoomRelation()
1816    local anotherPointData = mgr.bigCityMgr:GetRopewayPointDataByPoint(point)
1817    local mainEntityData = mapEntityData:GetMainEntityData()
1818    if anotherPointData then

1820    elseif mainEntityData then
1821        local anchorGridMap = mainEntityData.ropeWayAttr?.anchorGridMap
1822        for pointXY, _ in pairs(anchorGridMap or {}) do
1823            local gridX = pointXY >> 16
1824            local gridY = pointXY & 0xFFFF
1825            if gridX ~= mapEntityData.gridX or gridY ~= mapEntityData.gridY then
1826                anotherPointData = pointXY
1827            end
1828        end
1829        if anotherPointData then
1830            self:ShowRopewayPointLine(point,anotherPointData,relation,flag)
1831        end
1832    elseif not flag then
1833        for point1, v1 in pairs(self.linkedRopeLines) do
1834            for point2, _ in pairs(v1) do
1835                if point1 == point or point2 == point then
1836                    self:ShowRopewayPointLine(point1,point2,relation,flag)
1837                    return
1838                end
1839            end
1840        end
1841    elseif not mainEntityData  then
1842        self.cacheMainEntityRopeLine[mapEntityData:GetMainEntityId()] = {mapEntityData = mapEntityData,flag = flag}

1853local MapEffectGoRelationToLineColorType = {
1854    [RelationType.None]    = 4,
1855    [RelationType.Neutral] = 4,
1856    [RelationType.Self]    = 1,
1857    [RelationType.Enemy]   = 2,
1858    [RelationType.Friend]  = 3,
1859}
1860---@param point number
1861---@param anotherPoint number
1862---@param flag boolean
1863function cls:ShowRopewayPointLine(point,anotherPoint,relation,flag)
1864    if flag then
1865        if self:CheckHaveRopewayLine(point, anotherPoint) then
1866            return
1867        end
1868        local getWorldPos = function(pointXY)
1869            local gridX = pointXY >> 16
1870            local gridY = pointXY & 0xFFFF
1871            local worldPos = InfiniteZoomUtil.ConvertGridIndexToWorldPos(MapCoordUtil.ConvertGridToGridIndex(gridX or 0, gridY or 0))
1872            return worldPos
1873        end
1874        local startPos = getWorldPos(point)
1875        local endPos = getWorldPos(anotherPoint)
1876        local pointLine = {startPos, endPos}
1877        local colorType = MapEffectGoRelationToLineColorType[relation]
1878        local lineID = OutsideLuaCall.StaticCreateLine(pointLine, 0, colorType, 0, 0, 0, false, 8, 0.3, nil)
1879        if not self.linkedRopeLines[point] then
1880            self.linkedRopeLines[point] = {}
1881        end
1882        self.linkedRopeLines[point][anotherPoint] = lineID
1883    else
1884        local removeLineFun = function(point,anotherPoint)
1885            if self.linkedRopeLines[point] and self.linkedRopeLines[point][anotherPoint] then
1886                local lineID = self.linkedRopeLines[point][anotherPoint]
1887                OutsideLuaCall.StaticDeleteLine(lineID)
1888                self.linkedRopeLines[point][anotherPoint] = nil
1889                self.linkedRopeLines[point] = nil
1890                return true
1891            end
1892            return false
1893        end
1894        if not removeLineFun(point,anotherPoint) then
1895            removeLineFun(anotherPoint,point)
1896        end
1897    end
1898end

1900function cls:CheckHaveRopewayLine(point,anotherPoint)
1901    return (self.linkedRopeLines[point] and self.linkedRopeLines[point][anotherPoint]) or (self.linkedRopeLines[anotherPoint] and self.linkedRopeLines[anotherPoint][point])
1902end
