46function cls.GetMeshLineColor(entityId)
47    local entityData = mgr.mapCore.DataCtrl:GetDataByEntityId(entityId)
48    if not entityData then
49        if mgr.newbie?.NewbieMassiveArmyCtrl then
50            if mgr.newbie.NewbieMassiveArmyCtrl:IsMassiveArmyEntityId(entityId) then
51                return MeshLineUtil.MeshLineColor.Red
52            end
53        end
54
55        if mgr.civil.britainCivilCtrl.conquerBattleCtrl then
56            local color = mgr.civil.britainCivilCtrl.conquerBattleCtrl:GetLineColor(entityId)
57            if  color then
58                return color
59            end
60        end
61        return MeshLineUtil.MeshLineColor.White
62    end
63
64    return cls.GetMeshLineColorByEntityData(entityData)
65end

67-- 三方通知可以传入本次协议的状态，避免被尚未刷新的实体状态覆盖。
68function cls.GetMeshLineColorByEntityData(entityData, armyState)
69    local relation = mgr.outsideUtil.RelationShip.GetRelationByEntityData(entityData)
70    local relationColor = MeshLineUtil.MeshLineColorByRelation[relation]
71    if relation == RelationType.Self then
72        return MeshLineUtil.MeshLineColorByRelation[relation]
73    end
74
75    local getArmyState = entityData.GetArmyState
76    if armyState == nil and getArmyState ~= nil then
77        armyState = getArmyState(entityData)
78    end
79    if armyState == CfgNs.ArmyState.MS_Retreating then
80        return MeshLineUtil.MeshLineColor.White
81    end
82
83    local isFriendAlliance = WorldUtil.IsAllianceInDiplomacyData_EntityData(entityData)
84    if isFriendAlliance then
85        relationColor = MeshLineUtil.MeshLineColor.LightBlue
86    end
87
88    return relationColor or MeshLineUtil.MeshLineColor.Red
89end

91function cls.GetEntityMoveLineColor(fullUpdateLine, myAllianceId)
92    local entityData = mgr.mapCore.DataCtrl:GetDataByEntityId(fullUpdateLine.entityId)
93    if entityData then
94        return cls.GetMeshLineColorByEntityData(entityData, fullUpdateLine.armyState)
95    end
96
97    if fullUpdateLine.armyState == CfgNs.ArmyState.MS_Retreating then
98        return MeshLineUtil.MeshLineColor.White
99    end
100
101    local relation = RelationType.Enemy
102    if (fullUpdateLine.allianceId or 0) ~= 0 and myAllianceId ~= 0 and fullUpdateLine.allianceId == myAllianceId then
103        relation = RelationType.Friend
104    end
105
106    local colorType = MeshLineUtil.MeshLineColorByRelation[relation] or MeshLineUtil.MeshLineColor.White
107    if WorldUtil.IsAllianceInDiplomacyData(fullUpdateLine.allianceId) then
108        colorType = MeshLineUtil.MeshLineColor.LightBlue
109    end
110
111    return colorType
112end

119-- 头部：来源、ID、起止时间、颜色、地形开关、路点数、时间偏移数、特殊路点数。
120-- 后面依次为交错的 x/y、时间偏移、交错的特殊路点 x/y/z/w。
121function cls.CreateEntityMoveLineShared(param, enqueue)
122    -- 关闭后直接走原 table 传参入口，不写入或读取共享缓冲。
123    if not cls.EnableMoveLineSharedArray then
124        if enqueue then OutsideLuaCall.StaticEnqueueCreateEntityMoveLine(param)
125        else OutsideLuaCall.StaticCreateEntityMoveLine(param) end
126        return
127    end
128    local x, y, offsets = param.xArray, param.yArray, param.tickOffsets
129    local tx, ty, tz, tw = param.pathTypeXArray, param.pathTypeYArray, param.pathTypeZArray, param.pathTypeWArray
130    local entityId = math.tointeger(param.entityId or 0)
131    local startTick, endTick = math.tointeger(param.startTick or 0), math.tointeger(param.endTick or 0)
132    local count = x and #x or 0
133    local tickCount = offsets and #offsets or 0
134    local typeCount = tx and #tx or 0
135    if not entityId or not startTick or not endTick or count < 2 or not y or #y ~= count or
136        (tickCount > 0 and tickCount ~= count) or
137        (typeCount > 0 and (typeCount ~= count or not ty or #ty ~= count or not tz or #tz ~= count or not tw or #tw ~= count)) then
138        -- 异常参数沿用旧入口的校验及清理行为。
139        if enqueue then OutsideLuaCall.StaticEnqueueCreateEntityMoveLine(param)
140        else OutsideLuaCall.StaticCreateEntityMoveLine(param) end
141        return
142    end
143
144    local length = 9 + count * 2 + tickCount + typeCount * 4
145    if not s_moveLineSharedArray or s_moveLineSharedArray.length < length then
146        if s_moveLineSharedArray then s_moveLineSharedArray:CleanUp() end
147        s_moveLineSharedArray = LuaSharedArray.New(math.max(512, length), "EntityMoveLine")
148    end
149    local array = s_moveLineSharedArray
150    array[1], array[2] = param.moveLineType or 0, entityId
151    array[3], array[4] = startTick, endTick
152    array[5], array[6] = param.color or 0, param.autoCalTerrainHeight == true
153    array[7], array[8], array[9] = count, tickCount, typeCount
154    local index = 10
155    for i = 1, count do
156        array[index], array[index + 1] = x[i] or 0, y[i] or 0
157        index = index + 2
158    end
159    for i = 1, tickCount do
160        array[index] = offsets[i] or 0
161        index = index + 1
162    end
163    for i = 1, typeCount do
164        array[index], array[index + 1], array[index + 2], array[index + 3] = tx[i] or 0, ty[i] or 0, tz[i] or 0, tw[i] or 0
165        index = index + 4
166    end
167    -- C# 此次调用内完成快照，返回后即可覆盖此表；entityId/时间保持 Lua int64。
168    OutsideLuaCall.StaticCreateEntityMoveLineSharedArray(array:GetCSharpArray(), length, enqueue)
169end

180---@param fullUpdateLine NetNs.proto_YzEntityMoveLineAttr
181local function BuildEntityMoveLineCreateParam(fullUpdateLine, myAllianceId)
182    if not fullUpdateLine or not fullUpdateLine.pathPoint then
183        return nil
184    end
185
186    local pathPoint = fullUpdateLine.pathPoint
187    local coordPath = pathPoint.coordPath
188    if not coordPath then
189        return nil
190    end
191
192    local param = s_tempCreateParam
193    param.entityId = fullUpdateLine.entityId
194    param.xArray = coordPath.xArray
195    param.yArray = coordPath.yArray
196    param.tickOffsets = pathPoint.moveTickOffset
197    param.startTick = fullUpdateLine.startTick
198    param.endTick = fullUpdateLine.endTick
199    param.color = cls.GetEntityMoveLineColor(fullUpdateLine, myAllianceId)
200    -- 普通路径不传空表，避免每条线转换四个空 C# 数组；也防止沿用上一条特殊路径。
201    param.pathTypeXArray = nil
202    param.pathTypeYArray = nil
203    param.pathTypeZArray = nil
204    param.pathTypeWArray = nil
205
206    -- 普通路径直接返回，不遍历路点计数，也不清空八组特殊路径临时表。
207    local portalPathPoint = fullUpdateLine.portalPathPoint
208    if portalPathPoint and next(portalPathPoint) ~= nil then
209        local optimize = cls.EnableMoveLinePathPrepareOpt and pathPoint.xArray and pathPoint.yArray and
210            #pathPoint.xArray == #pathPoint.yArray
211        if not optimize then
212            TblUtil.Clear(s_tempPathPointsX)
213            TblUtil.Clear(s_tempPathPointsY)
214            TblUtil.Clear(s_tempPathTimeOffsets)
215            TblUtil.Clear(s_tempPointTypeXInfo)
216            TblUtil.Clear(s_tempPointTypeYInfo)
217            TblUtil.Clear(s_tempPointTypeZInfo)
218            TblUtil.Clear(s_tempPointTypeWInfo)
219        end
220        -- 特殊点记录含不同类型的字段，仍清空，避免复用旧记录的残留字段。
221        TblUtil.Clear(s_tempSpPointInfo)
222
223        local pathLength, timeLength
224        if optimize then
225            pathLength = pathPoint.xArray and #pathPoint.xArray or 0
226            timeLength = pathPoint.moveTickOffset and #pathPoint.moveTickOffset or 0
227        else
228            pathLength = TblUtil.GetTableLength(pathPoint.xArray)
229            timeLength = TblUtil.GetTableLength(pathPoint.moveTickOffset)
230        end
231        local performancePointCount = OutsideUtils.ProcessElevatorPoints(fullUpdateLine, s_tempSpPointInfo)
232        performancePointCount = OutsideUtils.ProcessRopewayPoints(fullUpdateLine, s_tempSpPointInfo, performancePointCount)
233        local lineX, lineY, lineTicks = s_tempPathPointsX, s_tempPathPointsY, s_tempPathTimeOffsets
234        local success, pathLengthAfterInsert
235        if optimize and performancePointCount == 0 and pathPoint.xArray and pathPoint.moveTickOffset then
236            -- 与插点函数的原始输入一致；仅省去无插点时的逐项复制，类型标记仍正常生成。
237            lineX, lineY, lineTicks = pathPoint.xArray, pathPoint.yArray, pathPoint.moveTickOffset
238            success, pathLengthAfterInsert = true, pathLength
239        else
240            success, pathLengthAfterInsert = OutsideUtils.ExecutePathPointInsertion(
241                fullUpdateLine, s_tempSpPointInfo, performancePointCount,
242                lineX, lineY, lineTicks, false)
243            if optimize and success then
244                TrimMoveLineArray(lineX, pathLengthAfterInsert)
245                TrimMoveLineArray(lineY, pathLengthAfterInsert)
246                TrimMoveLineArray(lineTicks, pathLength == timeLength and pathLengthAfterInsert or 0)
247            end
248        end
249
250        if success then
251            param.xArray = lineX
252            param.yArray = lineY
253            if pathLength == timeLength then
254                param.tickOffsets = lineTicks
255            end
256            OutsideUtils.ProcessPathTypes(
257                fullUpdateLine, s_tempSpPointInfo, performancePointCount,
258                lineX, lineY, nil, pathLengthAfterInsert, false,
259                s_tempPointTypeXInfo, s_tempPointTypeYInfo, s_tempPointTypeZInfo, s_tempPointTypeWInfo)
260            if optimize then
261                TrimMoveLineArray(s_tempPointTypeXInfo, pathLengthAfterInsert)
262                TrimMoveLineArray(s_tempPointTypeYInfo, pathLengthAfterInsert)
263                TrimMoveLineArray(s_tempPointTypeZInfo, pathLengthAfterInsert)
264                TrimMoveLineArray(s_tempPointTypeWInfo, pathLengthAfterInsert)
265            end
266            param.pathTypeXArray = s_tempPointTypeXInfo
267            param.pathTypeYArray = s_tempPointTypeYInfo
268            param.pathTypeZArray = s_tempPointTypeZInfo
269            param.pathTypeWArray = s_tempPointTypeWInfo
270        end
271    end
272    return param
273end
