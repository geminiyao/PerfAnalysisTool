455        public void UpdateStraightMoveLine()
456        {
457            bool enableOpt = ArmyEntityConst.EnableMoveLineRefreshOpt;
458            float thresholdSq = ArmyEntityConst.EntityMoveLineRefreshThreshold * ArmyEntityConst.EntityMoveLineRefreshThreshold;
459            foreach (var pair in m_straightLineEndPos)
460            {
461                long entityId = pair.Key;
462                var entity = MapManager.MapEntityManager.GetEntity(entityId);
463                if (GameCore.DotsMgr.EntityMgr.Exists(entity) == false)
464                {
465                    continue;
466                }
467
468                var startPos = GameCore.DotsMgr.EntityMgr.GetComponentData<Translation>(entity).Value;
469
470                if (enableOpt)
471                {
472                    if (m_straightLineLastStartPos.TryGetValue(entityId, out var lastPos) &&
473                        math.distancesq(startPos, lastPos) < thresholdSq)
474                    {
475                        continue;
476                    }
477                    m_straightLineLastStartPos[entityId] = startPos;
478                }
479
480                m_tempFloat3List.Clear();
481                m_tempFloat3List.Add(startPos);
482                m_tempFloat3List.Add(pair.Value);
483
484                RefreshArmyLine(entityId, (int) MeshLineDefine.ArmyLineIndexType.Move, m_tempFloat3List, ArmyEntityConst.ArmyLineSpeed);
485            }

1882        private void UpdateEntityMoveLine()
1883        {
1884            if (m_dictEntityMoveLineDatas.Count == 0) return;
1885
1886            bool enableOpt = ArmyEntityConst.EnableMoveLineRefreshOpt;
1887            int maxPerFrame = ArmyEntityConst.EntityMoveLineRefreshPerFrame;
1888            float thresholdSq = ArmyEntityConst.EntityMoveLineRefreshThreshold * ArmyEntityConst.EntityMoveLineRefreshThreshold;
1889
1890            int iterIndex = 0;
1891            int processedCount = 0;
1892            bool reachedStart = !enableOpt || m_entityMoveLineUpdateIndex == 0;
1893
1894            foreach (var pair in m_dictEntityMoveLineIds)
1895            {
1896                var moveLineIds = pair.Value;
1897                foreach (var pairLineId in moveLineIds)
1898                {
1899                    var lineId = pairLineId.Value;
1900                    if (!m_dictEntityMoveLineDatas.TryGetValue(lineId, out var moveLineData))
1901                        continue;
1902
1903                    iterIndex++;
1904                    if (!reachedStart)
1905                    {
1906                        if (iterIndex <= m_entityMoveLineUpdateIndex)
1907                            continue;
1908                        reachedStart = true;
1909                    }
1910
1911                    RefreshEntityMoveLineCurPos(ref moveLineData);
1912                    if (moveLineData.lineId > 0 && moveLineData.isPathValid)
1913                    {
1914                        bool needRefresh = !enableOpt || math.distancesq(moveLineData.curPos, moveLineData.lastRefreshedCurPos) > thresholdSq;
1915                        if (needRefresh)
1916                        {
1917                            if (ArmyEntityConst.MeshLine_UseStraightLineMode)
1918                            {
1919                                GameCore.MeshLineMgr.RefreshStraightLinePos((int) moveLineData.lineId, moveLineData.curPos, moveLineData.endPos);
1920                            }
1921                            else
1922                            {
1923                                GameCore.MeshLineMgr.RefreshLine((int) moveLineData.lineId, moveLineData.curPos, moveLineData.endPos, ArmyEntityConst.ArmyLineSpeed, moveLineData.color);
1924                            }
1925                            moveLineData.lastRefreshedCurPos = moveLineData.curPos;
1926                        }
1927                    }
1928                    m_dictEntityMoveLineDatas[lineId] = moveLineData;
1929
1930                    processedCount++;
1931                    if (enableOpt && processedCount >= maxPerFrame)
1932                    {
1933                        m_entityMoveLineUpdateIndex = iterIndex;
1934                        return;
1935                    }
1936                }
1937            }
1938            m_entityMoveLineUpdateIndex = 0;
1939        }
