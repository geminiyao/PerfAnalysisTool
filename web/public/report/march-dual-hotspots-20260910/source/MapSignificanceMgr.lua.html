1940function cls:AdjustMgrMaxUpdateTime()
1941    local isTaskOut = self._entitySignificanceTaskLinkedList:Length() > MapSignificanceDefine.MaxTaskOutCount
1942    if isTaskOut then
1943        local addTime = 0
1944        local canAdjust = (mgr.time.frame - self.lastNormalFrame) < AdjustInterval
1945        for type, maxTime in pairs(MgrMaxUpdateTime) do
1946            if type ~= TimeFrameType.BattleHeadMgr or canAdjust then
1947                addTime = addTime + maxTime
1948                self.mgrMaxUpdateTime[type] = 0
1949            else
1950                self.lastNormalFrame = mgr.time.frame
1951                self.mgrMaxUpdateTime[type] = maxTime
1952            end
1953        end
1954        self.mgrMaxUpdateTime[TimeFrameType.MapSignificanceMgr] = addTime
1955    else
1956        self.lastNormalFrame = mgr.time.frame
1957        for type, maxTime in pairs(MgrMaxUpdateTime) do
1958            self.mgrMaxUpdateTime[type] = maxTime
1959        end
1960    end
1961end
