286function cls:RefreshPresetTeam()
287    local teamDict = OutsideUtils.GetPresetTeamDatas(CfgNs.RaidPresetTeamType.RPTT_None)
288    if not teamDict then
289        return
290    end
291
292    -- 统计当前所有士兵数量
293    local leftSoldierNumDict = {}
294    local realSoldierNumDict = mgr.soldier:GetSoldiersDict() or EmptyTable
295    for k, v in pairs(realSoldierNumDict) do
296        leftSoldierNumDict[v.id] = v.inCityNumber
297    end
298
299    -- 减去所有部队身上士兵数量
300    for k, v in pairs(teamDict) do
301        if (v.soldierId or 0) ~= 0 and (v.soldierNum or 0) ~= 0 and not OutsideUtils.GetArmyDataByTeamDataInCludeMiracle(v) then
302            local count = leftSoldierNumDict[v.soldierId]
303            if count then
304                leftSoldierNumDict[v.soldierId] = math.max(0, count - v.soldierNum)
305            end
306        end
307    end
308
309    -- 统计需要自动补兵的队伍
310    local teamList = {}
311    for k, v in pairs(teamDict) do
312        if (v.soldierId or 0) ~= 0 and not OutsideUtils.GetArmyDataByTeamDataInCludeMiracle(v) then
313            local curNum = v.soldierNum or 0
314            local newNum = curNum
315            local marchLimit = OutsideUtils.GetTeamMarchLimitByHeroIDs(v.hero, v.loadedShipBerthId)
316            -- 处理自动补兵逻辑
317            if v.autoAddSoldierWhenReturn then
318                local leftNum = leftSoldierNumDict[v.soldierId] or 0
319                if leftNum > 0 then
320                    if curNum < marchLimit then
321                        local addNum = math.min(marchLimit - curNum, leftNum)
322                        newNum = curNum + addNum
323                        if curNum ~= newNum then
324                            leftSoldierNumDict[v.soldierId] = leftNum - addNum
325                        end
326                    end
327                end
328            end
329
330            -- 处理出征上限降低的逻辑（如个人落匪后）
331            if curNum == newNum then
332                if curNum > marchLimit then
333                    newNum = marchLimit
334                end
335            end
336
337            if curNum ~= newNum then
338                local result = 
339                {
340                    teamID = v.id,
341                    hero = v.hero,
342                    soldierId = v.soldierId,
343                    soldierNum = newNum,
344                    autoUseHeroSkill = v.autoUseHeroSkill,
345                    autoAddSoldierWhenReturn = v.autoAddSoldierWhenReturn,
346                    autoReturnHome = v.autoReturnHome,
347                }
348                table.insert(teamList, result)
349            end
350        end
351    end
352    if #teamList > 0 then
353        ---奇兵营逻辑
354        ---没有绑定精锐营，第五编队士兵归零
355        mgr.soldier.Data:SetFifthBarrackSoldierCount(teamList,Menu_ArmyMarch_MenuType.Outside)
356        mgr.outsideMap.MapNet:RequestBatchSetPresetTeamAsync(teamList, function(rspHead, rspBody)
357            local success = rspHead and rspHead.errorCode == 0
358            if success then
359                mgr.evt:Post(UIEvt.MenuMarchFloat_OnRefreshPresetTeam)
360            end
361        end)
362    end
363end

388function cls:IsTargetInArea(target,touchPos)
389    local scaleRatio = math.min(U3D.Screen.width, U3D.Screen.height) / 1080 -- 得到当前的屏幕缩放比
390
391    local screenPos = mgr.ui:GetUICamera():WorldToScreenPoint(target.transform.position)
392    screenPos = U3D.Vector2(screenPos.x, screenPos.y)
393    local offset = touchPos - screenPos
394    -- local text  = "touchx=%s,touchy=%s,=====targetx=%s,targety=%s,offsetx=%s,offsety=%s"
395    -- local text2 = string.format(text,touchPos.x,touchPos.y,screenPos.x,screenPos.y,offset.x,offset.y)
396    -- mgr.jasonwLog:Error(text2)
397    
398    local offsetX = math.abs(offset.x)
399    local offsetY = math.abs(offset.y)
400    local sizeX = target.rectTransform.rect.width * scaleRatio
401    local sizeY = target.rectTransform.rect.height * scaleRatio
402
403    if offsetX <= sizeX * 0.6 and offsetY <= sizeY * 0.6 then 
404        return true
405    end 
406
407    return false
408end

740function cls:GetRestraintRelationForBMYC(targetType,isBMYC)
741    if targetType == nil then
742        return RestraintRelationType.None
743    end
744    local restraint = BMYCRestraintRelationTbl[targetType]
745    if isBMYC == nil or isBMYC then
746        return restraint
747    elseif  restraint == RestraintRelationType.Proactive then
748        return RestraintRelationType.Passive
749    elseif  restraint == RestraintRelationType.Passive then
750        return RestraintRelationType.Proactive
751    end
752    return RestraintRelationType.None
753end

785function cls:RefreshRestraintIcon(selfEntityId, targetEntityId, isShow)
786    local selfArmyData = mgr.mapCore.DataCtrl:GetDataByEntityId(selfEntityId)
787    if not selfArmyData or selfArmyData.entityType ~= CfgNs.WorldEntityType.WET_Army then
788        return
789    end
790
791    ---@type Float_BattleLabelBubble
792    local uiBattleLabelBubble = mgr.ui:FindUI(UIName.Float_BattleLabelBubble)
793    if not uiBattleLabelBubble then return end
794
795    if isShow and targetEntityId > 0 then
796        if targetEntityId ~= self._restraintTargetId then
797            self._restraintTargetId = targetEntityId
798            TblUtil.Clear(self._preRestraintTypes)
799        end
800
801        local targetArmyData = mgr.mapCore.DataCtrl:GetDataByEntityId(targetEntityId)
802        -- 客户端创建对象也会走这里，先保护一下，targetArmyData可能为空
803        if not targetArmyData or targetArmyData.entityType ~= CfgNs.WorldEntityType.WET_Army then
804            return
805        end
806
807        local allSelectedArmyIdList = mgr.outsideArmy.ArmySelect:GetSelectedArmyIds()
808        local relation = mgr.outsideUtil.RelationShip.GetRelationByMapEntityData(targetArmyData)
809        local targetSoldierType, targetSoldierId = OutsideUtils.GetArmySoldierType(targetEntityId)
810        local mapCamera = mgr.mapVisual.CameraCtrl.camera
811        local targetSoldierBMYC = SoldierUtil.IsBMYC(targetSoldierId)
812
813        for _, myEntityId in ipairs (allSelectedArmyIdList) do
814            if relation == RelationType.Enemy then
815                -- UtilArmyProfile:Begin("PlayRestraintEffect")
816                local selfSoldierType, selfSoldierId = OutsideUtils.GetArmySoldierType(myEntityId)
817                local selfSoldierBMYC = SoldierUtil.IsBMYC(selfSoldierId)
818                local restraintRelationType = RestraintRelationType.None
819                if selfSoldierBMYC and targetSoldierBMYC  then
820                    restraintRelationType = RestraintRelationType.None
821                elseif selfSoldierBMYC then
822                    restraintRelationType = mgr.outsideArmy:GetRestraintRelationForBMYC({ targetSoldierType })
823                elseif targetSoldierBMYC then
824                    --目标方是白马义从是将克制关系反向处理
825                    restraintRelationType = mgr.outsideArmy:GetRestraintRelationForBMYC({ selfSoldierType })
826                    if restraintRelationType == RestraintRelationType.Proactive then
827                        restraintRelationType = RestraintRelationType.Passive
828                    elseif restraintRelationType == RestraintRelationType.Passive then
829                        restraintRelationType = RestraintRelationType.Proactive
830                    end
831                else
832                    restraintRelationType = self:GetRestraintRelation(selfSoldierType, targetSoldierType)
833                end
834                local preRetraintRelationType = self._preRestraintTypes[myEntityId]
835                if restraintRelationType ~= RestraintRelationType.None then
836                    if not (preRetraintRelationType ~= nil and preRetraintRelationType == restraintRelationType) then
837                        self._preRestraintTypes[myEntityId] = restraintRelationType
838                        -- UtilArmyProfile:Begin("ReleaseRestraintUI")
839                        self:ReleaseRestraintUI(myEntityId)
840                        -- UtilArmyProfile:End("ReleaseRestraintUI")
841                        
842                        -- UtilArmyProfile:Begin("StaticGetRestraintPositionInfo")
843                        local selfScreenPos, targetScreenPos, uiPos = ArmyEntityLuaCall.StaticGetRestraintPositionInfo(myEntityId, targetEntityId, mapCamera)
844                        -- UtilArmyProfile:End("StaticGetRestraintPositionInfo")
845                        -- UtilArmyProfile:Begin("RegisterRestraint")
846                        local restraintId, restraintUI = uiBattleLabelBubble:RegisterRestraint(uiPos)
847                        -- UtilArmyProfile:End("RegisterRestraint")
848
849                        if restraintId and restraintUI then
850                            self.restraintInfo[myEntityId] = restraintId
851                            local isLeft = selfScreenPos.x < targetScreenPos.x
852                            -- UtilArmyProfile:Begin("PlayRestraintAnim")
853                            self:PlayRestraintAnim(restraintUI, restraintRelationType, selfSoldierType, targetSoldierType, isLeft, selfSoldierId, targetSoldierId)
854                            -- UtilArmyProfile:End("PlayRestraintAnim")
855                        end
856                    end
857                else
858                    self._preRestraintTypes[myEntityId] = nil
859                    self:ReleaseRestraintUI(myEntityId)
860                end
861                -- UtilArmyProfile:End("PlayRestraintEffect")
862            else
863                self._preRestraintTypes[myEntityId] = nil
864                self:ReleaseRestraintUI(myEntityId)
865            end
866        end
867    else
868        self._restraintTargetId = 0
869        self:RealseAllRestraint()
870        TblUtil.Clear(self._preRestraintTypes)
871    end
872end

884local MaxMidPos = 4
885function cls:CalRestraintUIPos(selfArmyPos, targetArmyPos)
886    local midPos = (selfArmyPos + targetArmyPos) * 0.5
887    local distance = U3D.Vector3.Distance(selfArmyPos, targetArmyPos)
888    local uiWorldPos = midPos
889    if distance > MaxMidPos * 2 then
890        local dir = selfArmyPos - targetArmyPos
891        dir:Normalize()
892        uiWorldPos = targetArmyPos + dir * MaxMidPos
893    end
894
895    return uiWorldPos
896end

979function cls:OnCameraMove()
980    if not mgr.unlockData:IsUnlock(CfgNs.FeatureUnlockEnum.NFUE_BattleHeadOptimization) then
981        return
982    end
983
984    if mgr.raidMap.IsInRaid then return end
985
986    local worldPos = mgr.mapVisual.CameraCtrl.cameraCtrl.CorePos
987    if self.lastCorePos then
988        local sqrMagnitude = (self.lastCorePos - worldPos).sqrMagnitude

990            return
991        end
992    end
993
994    self.lastCorePos = worldPos
995
996    local curTime = mgr.time.realT
997    if curTime - (self.lastTime or 0) < 0.5 then
998        return
999    end
1000    
1001    self.lastTime = curTime
1002
1003    ProfilerUtil.BeginSample("OnCameraMove")
1004    local entityDict = mgr.mapCore.MapEntityCtrl.entitysByEntityId
1005    local buckets = mgr.battleHead.waitShowEntityQueue.buckets
1006    for i = 2, #buckets do -- 最高优先级不受摄像机位置影响
1007        for entityId, _ in pairs(buckets[i]) do
1008            (entityDict[entityId])?:RefreshDistancePriority()
1009        end
1010    end
1011
1012    buckets = mgr.battleHead.battleHeadQueue.buckets
1013    for i = 1, #buckets do 
1014        for entityId, _ in pairs(buckets[i]) do
1015            (entityDict[entityId])?:RefreshDistancePriority()
1016        end
1017    end
1018
1019    ProfilerUtil.EndSample()
1020end

1039function cls:RefreshAllArmyFactionColor()
1040    local isOutside = mgr.scene:IsInScene(SceneName.Outside)
1041    if not isOutside then
1042        return
1043    end
1044    if not mgr.outsideArmy or not mgr.outsideArmy.ArmyEntityCtrl then
1045        return
1046    end
1047    local allArmies = mgr.outsideArmy.ArmyEntityCtrl:GetAllArmyEntities()
1048    for entityId, armyEntity in pairs_safe(allArmies) do
1049        -- 友盟/联盟关系变化时，propertyOwner 属性不会 dirty，需主动刷新 3D 兵种染色
1050        if armyEntity and armyEntity.OnRefresh_uid_allianceId then
1051            armyEntity:OnRefresh_uid_allianceId()
1052            -- OnRefresh_uid_allianceId 已刷新 NameText/ArmyImg，这里只补 HeadImg
1053            mgr.evt:Post(OutsideEvt.BattleHeadItemRefresh, entityId, BattleHeadDefine.ItemType.HeadImg)
1054            goto continue
1055        end
1056        
1057        if armyEntity and armyEntity.UpdateRelationShip then
1058            armyEntity:UpdateRelationShip()
1059        end
1060        -- 最初逻辑：统一刷新 HeadImg/NameText（仅 entityId，不依赖 armyEntity 对象）
1061        mgr.evt:Post(OutsideEvt.BattleHeadItemRefresh, entityId, BattleHeadDefine.ItemType.HeadImg)
1062        mgr.evt:Post(OutsideEvt.BattleHeadItemRefresh, entityId, BattleHeadDefine.ItemType.NameText)
1063
1064        ::continue::
1065    end
1066end
