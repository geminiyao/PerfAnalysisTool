181 function cls:GetEntityByGridIndex(gridIndex)
182 local dict = self.entitysByGridIndex[gridIndex]
183 local length = TblUtil.GetTableLength(dict)
184 if length == 0 then
185 return nil
186 elseif length == 1 then
187 local k, v = next(dict)
188 return v
189 else
190 -- 排序优先级目前暂时由 entityType 决定：主要解决空地选中Entity覆盖地格原有Entity的问题
191 local list = TblUtil.GetTrimNilList(dict) or GlobalEmptyTable
192 table.sort(list, function (a, b)
193 return a.entityType > b.entityType
194 end)
195 return list[1]
196 end
197 end

199 --- 获取地格上可被选中的主实体。
200 --- 背景：部队驻守营帐等场景，客户端会在同一地格上额外创建 WET_YzGridEntity(309, 地格信息展示实体)。
201 --- 该实体不可选中(见 MapEntityOp:_canSingleSelect)，但其 entityType(309) 高于营帐(WET_Camp=308)，
202 --- 会被 GetEntityByGridIndex 的 entityType 降序排序误选为首选，导致跳转选中失败。
203 --- 这里优先返回可选中的真实实体(建筑/部队)，仅当地格上只有辅助地格实体时才退回原逻辑。
204 function cls:GetSelectableEntityByGridIndex(gridIndex)
205 local dict = self.entitysByGridIndex[gridIndex]
206 if not dict then
207 return nil
208 end
209 local list = TblUtil.GetTrimNilList(dict) or GlobalEmptyTable
210 if #list == 0 then
211 return nil
212 end
213 local filtered = {}
214 for _, v in ipairs(list) do
215 if v.entityType ~= CfgNs.WorldEntityType.WET_YzGridEntity then
216 table.insert(filtered, v)
217 end
218 end
219 if #filtered == 0 then
220 -- 地格上只有辅助地格实体，退回原逻辑
221 return self:GetEntityByGridIndex(gridIndex)
222 end
223 table.sort(filtered, function(a, b)
224 return a.entityType > b.entityType
225 end)
226 return filtered[1]
227 end

448 ---@param entityId number
449 ---@param delayed boolean 是否延迟销毁
450 ---@param deleteInfo NetNs.WorldEntityDeleteInfo | LocalEntityDeleteInfo (客户端自定义的本地Entity移除reason)
451 function cls:RemoveEntity(entityId, delayed, deleteInfo)
452 local mapEntity = self.entitysByEntityId[entityId]
453 if not mapEntity then
454 -- mgr.log:Error("RemoveEntity entityId failed : %s", entityId)
455 return
456 end
457
458 -- 防御：实体已被 CleanUp（如 ReleaseAllEntities 遍历期间，其它实体 RecycleViewGo
459 -- 反向调用本接口移除关联实体时，关联实体可能已先一步被 CleanUp，mapEntityData 已置 nil，
460 -- 但还在 entitysByEntityId 容器里）。此时只需清理容器引用即可，避免重复 CleanUp 与下方逻辑空指针。
461 if not mapEntity.mapEntityData then
462 self.entitysByEntityId[entityId] = nil
463 return
464 end
465
466 local oldGridIndex = mapEntity.mapEntityData.gridIndex
467 local newGridIndex = oldGridIndex
468 local entityType = mapEntity.entityType
469
470 if not newGridIndex then
471 newGridIndex = MapCoordUtil.ConvertGridToGridIndex(mapEntity.mapEntityData.gridX, mapEntity.mapEntityData.gridY)
472 end
473 --这里注意下时序：
474 -- step1 callback(entity)
475 -- step2 entity:CleanUp （CleanUp逻辑中可能会包含容器查找需求，所以需要在清空容器前处理）
476 -- step3 清空容器
477 if self.callbackOnDelEntity then
478 self.callbackOnDelEntity(mapEntity)
479 end
480
481 self:RemoveEntityInTypeData(mapEntity)
482 if not delayed then
483 --#if ENABLE_PROFILER
484 -- local sampler = self:GetSampler_RemoveMapEntity(entityType)
485 -- ProfilerUtil.SamplerBegin(sampler)
486 --#endif
487 mapEntity:CleanUp(deleteInfo)
488 --#if ENABLE_PROFILER
489 -- ProfilerUtil.SamplerEnd(sampler)
490 --#endif
491 end
492
493 if UtilArmyEntity.IsArmy(entityType) then
494 mgr.outsideArmy.ArmyEntityCtrl:ReleaseArmyEntity(entityId, newGridIndex)
495 end
496
497 if oldGridIndex then
498 local dict = self.entitysByGridIndex[oldGridIndex]
499 if dict then
500 dict[entityId] = nil
501 end
502 end
503
504 self.entitysByEntityId[entityId] = nil
505 end

574 function cls:RegisterFocusGridIndex(gridIndex, callback, allowFavoriteEmpty)
575 if gridIndex == nil or callback == nil then
576 return
577 end
578 local entityDict = self.entitysByGridIndex[gridIndex]
579 local callbackFired = false
580 if entityDict then
581 local createdEntityDict = self.entitysByGridIndex[gridIndex]
582 for k, v in pairs_safe(createdEntityDict) do
583 if allowFavoriteEmpty == true or v.entityType ~= CfgNs.WorldEntityType.WET_FavoriteEmpty then
584 callback(gridIndex, k)
585 callbackFired = true
586 end
587 end
588 end
589 --如果 callbackFired 为 false（entityDict 为空 table 或所有实体都被过滤），则 fallback 到注册监听的逻辑（检查本地资源田 / 注册 focusGridCallback）
590 if not callbackFired then
591 -- 检查是否有本地资源田视图但实体未创建
592 local resourceConfId = nil
593 if mgr.mapCore.LocalEntityMgr and mgr.mapCore.LocalEntityMgr.LocalResEntityHandler then
594 resourceConfId = mgr.mapCore.LocalEntityMgr.LocalResEntityHandler:TryGetLocalResourceConfId(gridIndex)
595 end
596
597 if resourceConfId then
598 -- 如果有本地资源田视图，触发实体化流程
599 local entityId = mgr.mapCore.LocalEntityMgr.LocalResEntityHandler:AddLocalResEntity(gridIndex)
600 if entityId then
601 -- 实体创建成功后，重新获取实体对象进行检查
602 local createdEntityDict = self.entitysByGridIndex[gridIndex]
603 for k, v in pairs_safe(createdEntityDict) do
604 callback(gridIndex, k)
605 end
606 else
607 -- 实体创建失败，使用原有的监听机制
608 if self.focusGridCallback[gridIndex] == nil then
609 self.focusGridCallback[gridIndex] = {}
610 end
611 table.insert(self.focusGridCallback[gridIndex], {allowFavoriteEmpty, callback})
612 end
613 else
614 -- 没有本地资源田视图，使用原有的监听机制
615 if self.focusGridCallback[gridIndex] == nil then
616 self.focusGridCallback[gridIndex] = {}
617 end
618 table.insert(self.focusGridCallback[gridIndex], {allowFavoriteEmpty, callback})
619 end
620 end
621 end
