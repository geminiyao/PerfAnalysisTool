109    public void CheckInit(bool forceInit = false)
110    {
111        // 仅判断 Inited 不够：[ExecuteInEditMode] 下经历编辑器域重载(Domain Reload)后，
112        // 组件实例会保留、但非序列化的运行时对象(camGroundFrustum / quadDriver / viewChecker 等)会被置空，
113        // 此时 Inited 可能仍为 true，导致 LateUpdate 直接空引用。用 camGroundFrustum 作为“运行时对象是否存活”的哨兵，
114        // 一旦丢失就强制重新初始化(这些对象都在本方法里一起创建)。
115        if (!forceInit && Inited && camGroundFrustum != null)
116        {
117            return;
118        }
119        
120        vtRenderer = TerrainVT_Manager.GetVTRenderer();
121        NodeCountNeedForDepth = new int[vtRenderer.MaxTreeDepth + 1];
122
123        CurrentTreeDepth = 0;
124        
125        lastCamPosition = Vector3.negativeInfinity;
126
127        singleTileTerrainSize = TerrainSize / TileCount;
128        singleTileRootSize = 1 << vtRenderer.MaxTreeDepth;
129        rootSize = singleTileRootSize * TileCount;
130        terrainSizeRcp = 1.0f / TerrainSize;
131
132        // IndexRT格式：R8（每像素1字节），编码方案：
133        //   低7位存 physicIndex（取值范围 [0, 127]，当前 PageCapacity 最大 64，足够）
134        //   高1位存 availableMip（0 或 1，当 VTRenderer.VTMipCount=1 时，availableMip∈{0,1}）
135        // nodeParam.x / nodeParam.z / nodeParam.size 不再存入 indexRT，Shader 侧通过
136        //   _VT_TerrainTileInfo.b (CurrentTreeDepth) + _VT_TerrainTileInfo.g (MaxTreeDepth) 重算
137        indexRT = new RenderTexture(rootSize, rootSize, 0, RenderTextureFormat.R8, RenderTextureReadWrite.Linear)
138        {
139            name = "IndexRT",
140            hideFlags = HideFlags.DontSave,
141            useMipMap = false,
142            wrapMode = TextureWrapMode.Clamp,
143            autoGenerateMips = false,
144            filterMode = FilterMode.Point
145        };
146        indexRT.Create();
147
148        // 每个脏结点占一个实例，同帧脏结点数不会超过 PageCapacity（每个都要占一个 physicIndex）
149        indexWriter = new IndexInstancedWriter(vtRenderer.FullScreenMesh, indexRT, rootSize, vtRenderer.PageCapacity);
150        viewChecker = new ViewChecker();
151        physicIndexManager = new PhysicIndexManager(vtRenderer.PageCapacity);
152        quadDriver = new QuadDriver(vtRenderer.PageCapacity);
153        camGroundFrustum = new CamGroundFrustum(WorldPosToTerrainSpaceXZ01, maxViewDistance, ViewEnlarge);
154
155        UpdateVTMode();
156
157        needRefreshTiles = true;
158
159        for (int tileX = 0; tileX < TileCount; ++tileX)
160        {
161            for (int tileY = 0; tileY < TileCount; ++tileY)
162            {
163                var tileIndex = ConvertTileIndex(tileX, ConvertTileY(tileY));
164                var tileName = $"Tile_{tileX:00}_{tileY:00}";
165                dictTileNameCache[tileIndex] = tileName;
166                dictTileNameCacheInvert[tileName] = tileIndex;
167            }
168        }
169
170        Inited = true;
171    }

190    private int CalcDesiredTreeDepth()
191    {
192        var groundPos = camGroundFrustum.frustumGroundPos;
193        var groundQuarterLength = (groundPos.rightQuarter_WS - groundPos.leftQuarter_WS).magnitude;
194
195        var gameRTSize = CalcGameRTSizeForVT(TerrainCamera);
196
197        float gamePixelPerMeter = gameRTSize.x / groundQuarterLength;
198        var pixelsRealNeeded = gamePixelPerMeter * singleTileTerrainSize;
199
200        var depthFloat = Mathf.Log(pixelsRealNeeded / VTRenderer.PageSize, 2);
201        int depth = Mathf.RoundToInt(depthFloat);
202
203        depth = Mathf.Clamp(depth, vtRenderer.MinTreeDepth, vtRenderer.MaxTreeDepth);
204        
205#if UNITY_EDITOR
206        var pixelDensityUsing = (VTRenderer.PageSize << depth) / singleTileTerrainSize;
207        PixelDensity = $"{gamePixelPerMeter}  {pixelDensityUsing}  Depth: {depth} / {depthFloat:N2}";
208#endif
209
210        return depth;
211    }

228    public static int NodeCountExtraBudget = 4; // Mip0（VT最精细Mip，每帧可以绘制的Page数量）
229    private void UpdateTreeDepth(int desiredTreeDepth)
230    {
231        int nextUseTreeDepth = desiredTreeDepth;
232        while (nextUseTreeDepth > vtRenderer.MinTreeDepth) // 找到结点数量<足够少>的最大的层
233        {
234            var needCount = NodeCountNeedForDepth[nextUseTreeDepth];
235            if (needCount > 0 && needCount <= vtRenderer.PageCapacity - NodeCountExtraBudget)
236            {
237                break;
238            }
239
240            --nextUseTreeDepth;
241        }
242
243        CurrentTreeDepth = Mathf.Clamp(nextUseTreeDepth, vtRenderer.MinTreeDepth, vtRenderer.MaxTreeDepth);
244    }

476    {
477        if (!PerfUpdateEnabled) return;
478        if (!IsVTActive) return;
479        if (TerrainCamera == null) return;
480
481        CalculateTerrainOffset();
482        CheckInit(false);
483        SyncKeepDetailOverride();
484
485        bool shouldRecalcNodeCountNeedForDepth = needRefreshTiles;
486        Profiler.BeginSample("[VT] RefreshTiles");
487        RefreshTiles(false);
488        Profiler.EndSample();
489
490        var camTransform = TerrainCamera.transform;
491        var camPosInTerrainSpace = WorldPosToTerrainSpace(camTransform.position);
492        if (CameraCtrl != null)
493        {
494            camPosInTerrainSpace.y -= CameraCtrl.GetCorePlaneY();
495        }
496
497        UpdateVTMode();
498        CheckDebugView();
499
500        Profiler.BeginSample("[VT] UpdateView");
501        if (forceViewCheck || CheckCameraMoved(TerrainCamera.transform.position))

503            UpdateViewChecker();

505            shouldRecalcNodeCountNeedForDepth = true;
506        }
507        Profiler.EndSample();
508

510        quadTreeDistanceComparer.frustumCenter = camGroundFrustum.frustumGroundPos.center_TS01;
511        listQuadTree.Sort(quadTreeDistanceComparer);
512
513        {

523        Profiler.EndSample();
524

539        {

542            ForceRedraw();
543        }
544        Profiler.EndSample();
545
546        // 在 RefreshTiles 之后、UpdateByViewRect 之前冲刷：新地块已建树，本帧 RedrawAllMips 能被立刻绘制
547        ProcessPendingMeshInfoOps();
548
549#if ENABLE_PROFILER
550        if (GMRenderPassToggle.IsVTSubmitEnabled())
551        {
552#endif
553        Profiler.BeginSample("[VT] UpdateByViewRect");
554        quadDriver.UpdateByViewRect(CurrentTreeDepth, listQuadTree, viewChecker);
555        Profiler.EndSample();
556
557        // 本帧攒下的所有 page 绘制与 IndexRT 索引写入，在这里一次性提交给 GPU。

559        vtRenderer.FlushRenderCommands();

563#if ENABLE_PROFILER
564        }
565#endif
566

568        int usingPageCount = vtRenderer.PageCapacity - physicIndexManager.FreeCount;
569        MaxUsedPageCount = usingPageCount > MaxUsedPageCount ? usingPageCount : MaxUsedPageCount;
570        PageUsageInfo = $"【{CurrentTreeDepth}】【{MaxUsedPageCount}】 【{usingPageCount} / {vtRenderer.PageCapacity}:{camPosInTerrainSpace.y:N2}】";
571
572        if (RealtimeUpdateInEditor)
573        {
574            NeedToForceRedraw = true;
575        }
576#endif

578
579#if UNITY_EDITOR && LINK_TEST

581        {
582            // Debug.Log($"VT.renderCounter 【{Time.frameCount}】 {renderCounterThisFrameForAllMip}  {renderDynamicMeterConsumedThisFrame}");
583        }

680    // 初始化Tile信息，或者镜头移动，导致Tile信息刷新（包括ListQuadTree的构建）
681    private void RefreshTiles(bool force)
682    {
683        if (!force && !needRefreshTiles)
684        {
685            return;
686        }
687        needRefreshTiles = false;
688
689        // 刷新地块信息列表
690        tileIndexToAddQuadTree.Clear();
691        tileIndexToDelQuadTree.Clear();
692
693        { // 刷新材质与Mesh信息
694            UpdateTileTransformDict(transform);
695
696            // 删除过时的
697            foreach (var kv in dictTerrainMeshMat)
698            {
699                var tileIndex = kv.Key;
700                if (!dictTileTransform.ContainsKey(tileIndex))
701                {
702                    tileIndexToDelQuadTree.Add(tileIndex);
703                }
704            }
705            
706            foreach (var tileIndex in tileIndexToDelQuadTree)
707            {
708                dictTerrainMeshMat.Remove(tileIndex);
709            }
710
711            // 添加新增的
712            foreach (var kv in dictTileTransform)
713            {
714                var tileIndex = kv.Key;
715                ConvertTileIndex(tileIndex, out var tileX, out var tileY);
716
717                var tileTransform = kv.Value;
718                if (tileTransform != null && !dictTerrainMeshMat.ContainsKey(tileIndex))
719                {
720                    var meshMatInfo = CreateSingleMaterialMesh(tileX, tileY, tileTransform.gameObject);
721                    if (meshMatInfo != null)
722                    {
723                        dictTerrainMeshMat.Add(tileIndex, meshMatInfo);
724                        tileIndexToAddQuadTree.Add(tileIndex);
725                    }
726                }
727            }
728
729            listTerrainMeshMat.Clear();
730            listTerrainMeshMat.AddRange(dictTerrainMeshMat.Values);
731        }
732
733        { // 刷新四叉树列表
734            for (int idx = listQuadTree.Count - 1; idx >= 0; --idx)
735            {
736                var quadTree = listQuadTree[idx];
737                if (tileIndexToDelQuadTree.Contains(quadTree.TileInfo.TileIndex))
738                {
739                    quadDriver.ReleaseQuadTree(quadTree);
740                    listQuadTree.RemoveAt(idx);
741                }
742            }
743
744            foreach (var tileIndex in tileIndexToAddQuadTree)
745            {
746                var info = dictTerrainMeshMat[tileIndex];
747                var quadTree = quadDriver.FetchQuadTree(this, rootSize, singleTileRootSize, info.tileInfo, physicIndexManager);
748                listQuadTree.Add(quadTree);
749            }
750        }
751
752        CalcValidArea();
753        
754#if UNITY_EDITOR && LINK_TEST
755        CheckMaterialCompatibility();
756#endif
757    }

759    private Vector4 validTileArea = new Vector4(0, 0, 1, 1); // xyzw 分别对应 xMin yMin xMax yMax
760    private void CalcValidArea()
761    {
762        if (listQuadTree.Count == 0)
763        {
764            validTileArea = new Vector4(0, 0, 1, 1);
765            return;
766        }
767
768        var firstArea = listQuadTree[0].treeArea;
769        validTileArea = firstArea;
770        for (int idx = 1; idx < listQuadTree.Count; ++idx)
771        {
772            var quadTree = listQuadTree[idx];
773            var area = quadTree.treeArea;
774            validTileArea.x = validTileArea.x < area.x ? validTileArea.x : area.x;
775            validTileArea.z = validTileArea.z > area.z ? validTileArea.z : area.z; 
776            validTileArea.y = validTileArea.y < area.y ? validTileArea.y : area.y; 
777            validTileArea.w = validTileArea.w > area.w ? validTileArea.w : area.w; 
778        }
779    }

799    public const int DefaultRenderLimitForMipZero = 16;
800    private const int DynamicMeterCostMipZero = 4;
801    private const int DynamicMeterCostNonMipZero = 1;
802    public static int RenderLimitForMipZero = VTRenderer.VTMipCount > 1

804        : System.Int32.MaxValue;

826    public void RenderQuadNode(QuadNode quadNode)
827    {
828#if ENABLE_PROFILER
829        if (!GMRenderPassToggle.IsVTSubmitEnabled())
830        {
831            return;
832        }
833#endif
834
835        if (quadNode.physicTexIndex <= InvalidPhysicIndex)
836        {
837#if UNITY_EDITOR && LINK_TEST
838            Debug.LogError($"Attempt to Render QuadeNode with physicTexIndex {quadNode.physicTexIndex}");
839#endif
840            return;
841        }
842
843        bool isPendingTextureMipmapRedraw = quadNode.HasPendingTextureMipmapRedraw;
844        if (quadNode.availableMip <= 0 && !isPendingTextureMipmapRedraw) // 已经绘制完成所有mip
845        {
846            return;
847        }
848
849        // 绘制 Mip0 受本帧动态计量约束：剩余 < Mip0消耗则拒绝；非Mip0始终绘制
850        int renderingMip = isPendingTextureMipmapRedraw
851            ? quadNode.pendingTextureMipmapRedrawMip
852            : quadNode.availableMip - 1;
853        if (renderingMip == 0 &&
854            RenderLimitForMipZero - renderDynamicMeterConsumedThisFrame < DynamicMeterCostMipZero)
855        {
856            return;
857        }
858
859        var patchInfo = quadNode.GetUVInSingleTile();
860        var renderTask = new VTRenderer.RenderTask(quadNode.physicTexIndex, renderingMip,
861            quadNode.tileInfo, patchInfo, quadNode.nodeParam.nodeArea, quadNode.hashMeshInfoIndex);
862        bool textureMipsReady = vtRenderer.Render(this, renderTask);
863
864        if (!isPendingTextureMipmapRedraw)
865        {
866            quadNode.availableMip = renderingMip;
867        }
868        quadNode.pendingTextureMipmapRedrawMip = textureMipsReady ? -1 : renderingMip;
869
870        var nodeParam = quadNode.nodeParam;
871        indexWriter.FillSubRegion(quadNode.IndexByteInfo, nodeParam.x, nodeParam.z, nodeParam.size);
872
873        ++renderCounterThisFrameForAllMip;
874        renderDynamicMeterConsumedThisFrame += renderingMip == 0
875            ? DynamicMeterCostMipZero
876            : DynamicMeterCostNonMipZero;
877        
878#if UNITY_EDITOR && LINK_TEST
879        // Debug.LogWarning($"VT.RenderQuadNode 【{Time.frameCount}】 {renderCounter}");
880#endif
881    }

962    private struct PendingMeshInfoOp
963    {
964        public int meshInfoIndex;
965        public MeshInfo meshInfo;
966        public MeshInfoOp op;
967
968        public PendingMeshInfoOp(int meshInfoIndex, MeshInfo meshInfo, MeshInfoOp op)
969        {
970            this.meshInfoIndex = meshInfoIndex;
971            this.meshInfo = meshInfo;
972            this.op = op;
973        }
974    }

976    private readonly Queue<PendingMeshInfoOp> pendingMeshInfoOps = new Queue<PendingMeshInfoOp>(256);
977    private readonly HashSet<int> pendingAddMeshInfoIndices = new HashSet<int>();
978
979    public int RegisterMeshInfo(MeshInfo meshInfo)
980    {
981        if (!HardwareSupportsDecal || meshInfo == null) return -1;
982
983        int meshInfoIndex = ++nextMeshInfoIndex;
984        dictMeshInfo[meshInfoIndex] = meshInfo;
985        pendingAddMeshInfoIndices.Add(meshInfoIndex);
986        pendingMeshInfoOps.Enqueue(new PendingMeshInfoOp(meshInfoIndex, meshInfo, MeshInfoOp.Add));
987        return meshInfoIndex;
988    }
989
990    public void DeregisterMeshInfo(int meshInfoIndex)
991    {
992        if (dictMeshInfo == null) return;
993
994        if (!dictMeshInfo.TryGetValue(meshInfoIndex, out var meshInfo))
995        {
996            return;
997        }
998
999        dictMeshInfo.Remove(meshInfoIndex);
1000
1001        // Add 还在队列里：直接取消，避免先插入再删除
1002        if (pendingAddMeshInfoIndices.Remove(meshInfoIndex))
1003        {
1004            return;
1005        }
1006
1007        if (listQuadTree == null) return;
1008        pendingMeshInfoOps.Enqueue(new PendingMeshInfoOp(meshInfoIndex, meshInfo, MeshInfoOp.Del));
1009    }

1011    private void ProcessPendingMeshInfoOps()
1012    {
1013        int count = pendingMeshInfoOps.Count;
1014        if (count == 0) return;
1015
1016        int budget = MaxMeshInfoOpsPerFrame;
1017        if (budget <= 0 || budget > count) budget = count;
1018
1019        Profiler.BeginSample("[VT] PendingMeshInfo");
1020        int processed = 0;
1021        while (processed < budget && pendingMeshInfoOps.Count > 0)
1022        {
1023            var op = pendingMeshInfoOps.Dequeue();
1024            if (op.op == MeshInfoOp.Add && !pendingAddMeshInfoIndices.Remove(op.meshInfoIndex))
1025            {
1026                continue;
1027            }
1028
1029            ApplyMeshInfoOp(op.meshInfoIndex, op.meshInfo, op.op);
1030            ++processed;
1031        }
1032        Profiler.EndSample();
1033    }

1035    private void ApplyMeshInfoOp(int meshInfoIndex, MeshInfo meshInfo, MeshInfoOp meshInfoOp)
1036    {
1037        if (meshInfo == null || listQuadTree == null) return;
1038
1039        var meshArea = WorldBoundsToTerrainArea(meshInfo.boundsInfo);
1040        foreach (var quadTree in listQuadTree)
1041        {
1042            quadTree.AddOrDelMeshInfo(meshInfoIndex, meshInfo, meshArea, meshInfoOp);
1043        }
1044    }

1147    /// 按包边石列表的 XZ 包围盒并集，只重绘覆盖到的 VT page。
1148    /// 用注册时缓存的 unionBounds，不要逐项重算：list 里的 mesh 可能已被 MeshLODInfo.Clear() 置空，
1149    /// 重算会因为 isValid 全 false 而退化成空操作，恰好漏掉最需要失效的那批 page。
1150    /// 这里的失效范围也因此与 DrawCliffWorldY 的粗筛范围严格一致（两者读同一份 unionBounds）。
1151    private void RedrawVTMeshInfoList(VTMeshInfoList meshInfoList)
1152    {
1153        if (meshInfoList?.listMeshInfo == null || !meshInfoList.hasUnionBounds || listQuadTree == null) return;
1154
1155        // RedrawByMeshInfo 只用 meshInfo 判空和读 isVisible，取任意一项即可
1156        MeshInfo dummy = null;
1157        var list = meshInfoList.listMeshInfo;
1158        for (int i = 0; i < list.Count; ++i)
1159        {
1160            if (list[i] != null)
1161            {
1162                dummy = list[i];
1163                break;
1164            }
1165        }
1166
1167        if (dummy == null) return;
1168
1169        var meshArea = WorldBoundsToTerrainArea(meshInfoList.unionBounds);
1170        foreach (var quadTree in listQuadTree)
1171        {
1172            quadTree.RedrawByMeshInfo(dummy, meshArea, MeshInfoOp.Add);
1173        }
1174    }
