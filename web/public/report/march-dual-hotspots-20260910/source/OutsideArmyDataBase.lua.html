239function cls:GetBattleCenterPosAndDir()
240    local pos, dir
241    local positionAttr = self.positionAttr
242    if positionAttr?.position then
243        pos = MapCoordUtil.ConvertProtoPosToWorldPos(positionAttr.position)
244    else
245        pos = MapCoordUtil.ConvertGridToWorldPos(MapCoordUtil.ConvertWorldPosToGrid(OutsideUtils.GetArmyPos(self.entityId)))
246    end
247
248    if positionAttr?.yaw then
249        dir = U3D.Vector3(positionAttr.yaw.x or 0, 0, -1 * (positionAttr.yaw.y or 0))
250        dir = dir.normalized
251    else
252        dir = U3D.Vector3(1, 0, 0)
253    end
254    return pos, dir
255end

412function cls:GetFightWallTargetPos(isBattleOnWall)
413    local yzArmyStatesAttr = self.yzArmyStatesAttr
414    if yzArmyStatesAttr and yzArmyStatesAttr.wallFightPosSet then
415        local posX = yzArmyStatesAttr.wallFightDstPosX
416        local posY = yzArmyStatesAttr.wallFightDstPosY
417        if posX ~= nil and posY ~= nil then
418            local wallFightDstPos = MapCoordUtil.ConvertCellToWorldPos(posX, posY)
419
420            return wallFightDstPos
421        end
422    end
423
424    local centerPos, dir = self:GetBattleCenterPosAndDir()
425    local targetPos = nil
426    if isBattleOnWall then
427        targetPos = centerPos + dir * OutsideDataDef.UnitBlockWidth * 2.5
428    else
429        local dist = 0
430        local fightingAreaEntityType = self.combatAttr?.fightingAreaEntityType
431        if fightingAreaEntityType == CfgNs.WorldEntityType.WET_PlayerCity then
432            dist = 2
433        end
434
435        targetPos = centerPos + dir * OutsideDataDef.UnitBlockWidth * (dist + 0.25)
436    end
437
438    return targetPos
439end

441function cls:GetShootArrowTargetPos()
442    local centerPos, dir = self:GetBattleCenterPosAndDir()
443    centerPos = OutsideUtils.GetArmyPosLua(self.entityId) 
444
445    local targetPos = nil
446    if self.yzArmyStatesAttr?.shootingArrow then --墙上防守士兵朝墙下射箭
447        targetPos = centerPos + dir * OutsideDataDef.UnitBlockWidth * (1.5 + 0.2 + 1)
448        local gridX, gridY = MapCoordUtil.ConvertWorldPosToGrid(targetPos)
449        targetPos.y = MapCoordUtil.GetTerrainHeight(gridX, gridY)
450    else --墙下进攻士兵朝墙上射箭
451        -- 在扔火把目标点的基础上，往后挪一格
452        targetPos = centerPos + dir * OutsideDataDef.UnitBlockWidth * (1.5 + 0.2 + 1)
453        targetPos.y = 2.5 -- 射到城墙的高度先统一设一个值
454    end
455
456    return targetPos
457end
