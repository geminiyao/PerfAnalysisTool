62 [MenuItem("Tools/Perf/Raw to Pdata (Batchmode)")]
63 public static void Convert()
64 {
65 string inputPath = Environment.GetEnvironmentVariable("RAW_TO_PDATA_INPUT");
66 string outputPath = Environment.GetEnvironmentVariable("RAW_TO_PDATA_OUTPUT");
67 s_LogFile = Environment.GetEnvironmentVariable("RAW_TO_PDATA_LOG") ?? "";
68
69 try
70 {
71 Log("[Batchmode] RawToPdata starting");
72
73 if (string.IsNullOrEmpty(inputPath) || string.IsNullOrEmpty(outputPath))
74 {
75 Log("[Batchmode] ERROR: RAW_TO_PDATA_INPUT or RAW_TO_PDATA_OUTPUT not set");
76 EditorApplication.Exit(1);
77 return;
78 }
79
80 if (!File.Exists(inputPath))
81 {
82 Log("[Batchmode] ERROR: input file not found: " + inputPath);
83 EditorApplication.Exit(1);
84 return;
85 }
86
87 Log("[Batchmode] Input: " + inputPath);
88 Log("[Batchmode] Output: " + outputPath);
89
90 // 1. 加载 .raw 到 Profiler
91 Log("[Batchmode] Loading profile data...");
92 ProfilerDriver.LoadProfile(inputPath, false);
93
94 // 等待 Profiler 数据就绪 (最多等 30 秒)
95 int waited = 0;
96 while (ProfilerDriver.lastFrameIndex < 0 && waited < 300)
97 {
98 System.Threading.Thread.Sleep(100);
99 waited++;
100 }
101
102 int firstFrame = ProfilerDriver.firstFrameIndex;
103 int lastFrame = ProfilerDriver.lastFrameIndex;
104
105 if (lastFrame < firstFrame || lastFrame < 0)
106 {
107 Log("[Batchmode] ERROR: no frames available after load. firstFrame=" + firstFrame + " lastFrame=" + lastFrame);
108 EditorApplication.Exit(1);
109 return;
110 }
111
112 Log("[Batchmode] Frames: " + firstFrame + " to " + lastFrame + " (" + (lastFrame - firstFrame + 1) + " total)");
113
114 // 2. 提取 markers + 写 .pdata
115 var pdata = ExtractMarkers(firstFrame, lastFrame);
116 if (pdata.frames.Count == 0)
117 {
118 Log("[Batchmode] ERROR: no complete frames extracted");
119 EditorApplication.Exit(1);
120 return;
121 }
122
123 WritePdata(outputPath, pdata);
124 Log("[Batchmode] Wrote .pdata: " + pdata.frames.Count + " frames, " + pdata.markerNames.Count + " markers");
125
126 // 3. 提取 counters + 写 .counters.json
127 // 3a. 尝试初始化 ProfilerWindow（batchmode 下 GetStatisticsValues 可能需要它）
128 try
129 {
130 var pwType = System.Type.GetType("UnityEditor.ProfilerWindow,UnityEditor");
131 if (pwType != null)
132 {
133 var pw = EditorWindow.GetWindow(pwType);
134 Log("[Batchmode] ProfilerWindow initialized for counter extraction");
135 }
136 }
137 catch (Exception e) { Log("[Batchmode] ProfilerWindow init skipped: " + e.Message); }
138
139 string sidecarPath = Path.ChangeExtension(outputPath, ".counters.json");
140 int countersFirst = pdata.frameIndexOffset;
141 int countersLast = pdata.frameIndexOffset + pdata.frames.Count - 1;
142 ExportCounters(sidecarPath, countersFirst, countersLast, pdata.frameIndexOffset);
143 Log("[Batchmode] Wrote .counters.json: " + sidecarPath);
144
145 Log("[Batchmode] SUCCESS");
146 EditorApplication.Exit(0);
147 }
148 catch (Exception e)
149 {
150 Log("[Batchmode] EXCEPTION: " + e.ToString());
151 EditorApplication.Exit(1);
152 }
153 }

155 // 一次 Unity 启动批量转换 manifest 中的 raw，避免专项多 case 数据逐文件启动 Editor。
156 // RAW_TO_PDATA_MANIFEST: UTF-8 文本，每行一个 raw 绝对路径；pdata 写在 raw 同目录。
157 public static void ConvertManifest()
158 {
159 string manifestPath = Environment.GetEnvironmentVariable("RAW_TO_PDATA_MANIFEST");
160 s_LogFile = Environment.GetEnvironmentVariable("RAW_TO_PDATA_LOG") ?? "";
161 int failed = 0;
162 try
163 {
164 if (string.IsNullOrEmpty(manifestPath) || !File.Exists(manifestPath))
165 throw new FileNotFoundException("RAW_TO_PDATA_MANIFEST not found", manifestPath);
166
167 string[] inputs = File.ReadAllLines(manifestPath, Encoding.UTF8);
168 foreach (string rawLine in inputs)
169 {
170 string inputPath = rawLine.Trim();
171 if (string.IsNullOrEmpty(inputPath)) continue;
172 string outputPath = Path.ChangeExtension(inputPath, ".pdata");
173 try
174 {
175 ConvertOne(inputPath, outputPath);
176 }
177 catch (Exception e)
178 {
179 failed++;
180 Log("[Batchmode] FAILED: " + inputPath + "\n" + e);
181 }
182 }
183 Log("[Batchmode] Manifest complete, failed=" + failed);
184 EditorApplication.Exit(failed == 0 ? 0 : 1);
185 }
186 catch (Exception e)
187 {
188 Log("[Batchmode] MANIFEST EXCEPTION: " + e);
189 EditorApplication.Exit(1);
190 }
191 }

193 private static void ConvertOne(string inputPath, string outputPath)
194 {
195 if (!File.Exists(inputPath))
196 throw new FileNotFoundException("input raw not found", inputPath);
197
198 Log("[Batchmode] Input: " + inputPath);
199 ProfilerDriver.LoadProfile(inputPath, false);
200 int waited = 0;
201 while (ProfilerDriver.lastFrameIndex < 0 && waited < 300)
202 {
203 System.Threading.Thread.Sleep(100);
204 waited++;
205 }
206 int firstFrame = ProfilerDriver.firstFrameIndex;
207 int lastFrame = ProfilerDriver.lastFrameIndex;
208 if (lastFrame < firstFrame || lastFrame < 0)
209 throw new InvalidDataException("no frames after load: " + firstFrame + ".." + lastFrame);
210
211 var pdata = ExtractMarkers(firstFrame, lastFrame);
212 if (pdata.frames.Count == 0)
213 throw new InvalidDataException("no complete frames extracted");
214 WritePdata(outputPath, pdata);
215 Log("[Batchmode] Wrote .pdata: " + pdata.frames.Count + " frames, "
216 + pdata.markerNames.Count + " markers -> " + outputPath);
217
218 string sidecarPath = Path.ChangeExtension(outputPath, ".counters.json");
219 int countersFirst = pdata.frameIndexOffset;
220 int countersLast = pdata.frameIndexOffset + pdata.frames.Count - 1;
221 ExportCounters(sidecarPath, countersFirst, countersLast, pdata.frameIndexOffset);
222 }

226 private class MyMarker { public int nameIndex; public float msMarkerTotal; public int depth; }
227 private class MyThread { public int threadIndex; public List<MyMarker> markers = new List<MyMarker>(); }
228 private class MyFrame { public double msStartTime; public float msFrame; public List<MyThread> threads = new List<MyThread>(); }
229 private class MyProfileData
230 {
231 public int frameIndexOffset;
232 public List<MyFrame> frames = new List<MyFrame>();
233 public List<string> markerNames = new List<string>();
234 public List<string> threadNames = new List<string>();
235 public Dictionary<string, int> markerNameDict = new Dictionary<string, int>();
236 public Dictionary<string, int> threadNameDict = new Dictionary<string, int>();
237 }
238
239 private static MyProfileData ExtractMarkers(int firstFrame, int lastFrame)
240 {
241 var data = new MyProfileData();
242 data.frameIndexOffset = firstFrame;
243
244 var iter = new ProfilerFrameDataIterator();
245 try
246 {
247 int totalFrames = lastFrame - firstFrame + 1;
248 var threadNameCount = new Dictionary<string, int>();
249
250 for (int frameIndex = firstFrame; frameIndex <= lastFrame; frameIndex++)
251 {
252 int threadCount = iter.GetThreadCount(frameIndex);
253 iter.SetRoot(frameIndex, 0);
254 float msFrame = iter.frameTimeMS;
255
256 // 跳掉首尾 incomplete 帧
257 if ((frameIndex == firstFrame || frameIndex == lastFrame) &&
258 firstFrame != lastFrame && msFrame == 0)
259 {
260 if (frameIndex == firstFrame)
261 {
262 data.frameIndexOffset = frameIndex + 1;
263 continue;
264 }
265 else break;
266 }
267
268 var frame = new MyFrame
269 {
270 msStartTime = 1000.0 * iter.GetFrameStartS(frameIndex),
271 msFrame = msFrame
272 };
273 data.frames.Add(frame);
274
275 threadNameCount.Clear();
276 for (int threadIndex = 0; threadIndex < threadCount; threadIndex++)
277 {
278 iter.SetRoot(frameIndex, threadIndex);
279 string threadName = iter.GetThreadName();
280 if (string.IsNullOrEmpty(threadName) || threadName.Trim() == "") continue;
281 string groupName = iter.GetGroupName();
282 string fullThreadName = string.IsNullOrEmpty(groupName)
283 ? threadName
284 : string.Format("{0}.{1}", groupName, threadName);
285
286 var t = new MyThread();
287 frame.threads.Add(t);
288
289 int nameCount;
290 threadNameCount.TryGetValue(fullThreadName, out nameCount);
291 threadNameCount[fullThreadName] = nameCount + 1;
292 string threadNameWithIndex = string.Format("{0}:{1}", threadNameCount[fullThreadName], fullThreadName);
293 threadNameWithIndex = CorrectThreadName(threadNameWithIndex);
294 t.threadIndex = AddOrGetIndex(data.threadNames, data.threadNameDict, threadNameWithIndex);
295
296 const bool enterChildren = true;
297 while (iter.Next(enterChildren))
298 {
299 float duration = iter.durationMS;
300 if (duration < 0) continue;
301 var m = new MyMarker { msMarkerTotal = duration, depth = iter.depth };
302 m.nameIndex = AddOrGetIndex(data.markerNames, data.markerNameDict, iter.name);
303 t.markers.Add(m);
304 }
305 }
306
307 if ((frameIndex - firstFrame) % 50 == 0)
308 Log("[Batchmode] Extracting markers: frame " + (frameIndex - firstFrame + 1) + "/" + totalFrames);
309 }
310 }
311 finally { iter.Dispose(); }
312
313 return data;
314 }

316 private static int AddOrGetIndex(List<string> list, Dictionary<string, int> dict, string name)
317 {
318 int idx;
319 if (dict.TryGetValue(name, out idx)) return idx;
320 list.Add(name);
321 idx = list.Count - 1;
322 dict[name] = idx;
323 return idx;
324 }

326 private static string CorrectThreadName(string threadNameWithIndex)
327 {
328 var info = threadNameWithIndex.Split(':');
329 if (info.Length >= 2)
330 {
331 string idxStr = info[0];
332 string name = info[1];
333 if (name.Trim() == "")
334 {
335 threadNameWithIndex = string.Format("{0}:[Unknown]", idxStr);
336 }
337 else
338 {
339 var m = System.Text.RegularExpressions.Regex.Match(name, @"^(.*[^\s])\s+(\d+)$");
340 if (m.Success)
341 {
342 string prefix = m.Groups[1].Value;
343 int groupIdx = 1 + int.Parse(m.Groups[2].Value);
344 threadNameWithIndex = string.Format("{0}:{1}", groupIdx, prefix);
345 }
346 }
347 }
348 return threadNameWithIndex.Trim();
349 }

351 // ==================== .pdata 序列化 (schema v7, 与 ProfileData.cs 一致) ====================

353 private static void WritePdata(string path, MyProfileData data)
354 {
355 using (var fs = new FileStream(path, FileMode.OpenOrCreate, FileAccess.Write))
356 using (var bw = new BinaryWriter(fs))
357 {
358 bw.Write(PdataVersion); // int32 version
359 bw.Write(data.frameIndexOffset); // int32
360 bw.Write(data.frames.Count); // int32 frameCount
361 foreach (var frame in data.frames)
362 {
363 bw.Write(frame.msStartTime); // double
364 bw.Write(frame.msFrame); // float
365 bw.Write(frame.threads.Count); // int32
366 foreach (var t in frame.threads)
367 {
368 bw.Write(t.threadIndex); // int32
369 bw.Write(t.markers.Count); // int32
370 foreach (var m in t.markers)
371 {
372 bw.Write(m.nameIndex); // int32
373 bw.Write(m.msMarkerTotal); // float
374 bw.Write(m.depth); // int32
375 }
376 }
377 }
378 bw.Write(data.markerNames.Count); // int32
379 foreach (var name in data.markerNames) bw.Write(name); // length-prefixed
380 bw.Write(data.threadNames.Count); // int32
381 foreach (var name in data.threadNames) bw.Write(name); // length-prefixed
382 }
383 }

385 // ==================== Counters 提取 (2019.4 兜底路径) ====================
386
387 private static void ExportCounters(string sidecarPath, int firstFrame, int lastFrame, int frameIndexOffset)
388 {
389 int frameCount = lastFrame - firstFrame + 1;
390 if (frameCount <= 0) { File.WriteAllText(sidecarPath, EmptySidecarJson(frameIndexOffset, 0)); return; }
391
392 int N = SchemaFieldNames.Length;
393
394 // 收集 stat 名
395 var areaStats = new Dictionary<ProfilerArea, string[]>();
396 foreach (ProfilerArea area in System.Enum.GetValues(typeof(ProfilerArea)))
397 {
398 if ((int)area < 0) continue;
399 string[] stats = null;
400 try { stats = ProfilerDriver.GetGraphStatisticsPropertiesForArea(area); } catch { }
401 areaStats[area] = stats ?? new string[0];
402 }
403
404 // fuzzy match
405 var identifiers = new int[N];
406 var columns = new float[N][];
407 int validCount = 0;
408 for (int i = 0; i < N; i++)
409 {
410 var refDef = CounterStatRefs[i];
411 if (string.IsNullOrEmpty(refDef.statName)) { identifiers[i] = -1; columns[i] = null; continue; }
412 string[] candidates;
413 if (!areaStats.TryGetValue(refDef.area, out candidates) || candidates.Length == 0)
414 { identifiers[i] = -1; columns[i] = null; continue; }
415
416 string match = null;
417 foreach (var s in candidates) { if (string.Equals(s, refDef.statName, StringComparison.OrdinalIgnoreCase)) { match = s; break; } }
418 if (match == null) foreach (var s in candidates) { if (s.IndexOf(refDef.statName, StringComparison.OrdinalIgnoreCase) >= 0) { match = s; break; } }
419
420 if (match != null)
421 {
422 int id = ProfilerDriver.GetStatisticsIdentifierForArea(refDef.area, match);
423 if (id >= 0)
424 {
425 var buf = new float[frameCount];
426 float maxValue;
427 // Unity 2019.4: GetStatisticsValues 返回 void，数据写入 buf
428 ProfilerDriver.GetStatisticsValues(id, firstFrame, 1.0f, buf, out maxValue);
429 // 检查是否拿到了真实数据
430 bool hasNonZero = false;
431 int got = 0;
432 for (int j = 0; j < frameCount; j++) { if (buf[j] != 0f) { hasNonZero = true; got++; } }
433 Log("[Batchmode] Counter '" + refDef.statName + "' (area=" + refDef.area +
434 "): id=" + id + " got=" + got + "/" + frameCount + " maxVal=" + maxValue + " nonZero=" + hasNonZero);
435 identifiers[i] = id; columns[i] = buf; validCount++; continue;
436 }
437 }
438 identifiers[i] = -1; columns[i] = null;
439 }
440
441 Log("[Batchmode] Counters resolved: " + validCount + "/" + N);
442 if (validCount == 0) { File.WriteAllText(sidecarPath, EmptySidecarJson(frameIndexOffset, 0)); return; }
443
444 var sb = new StringBuilder(1024 * 64);
445 BeginSidecar(sb, frameIndexOffset);
446 int written = 0;
447 for (int row = 0; row < frameCount; row++)
448 {
449 int frameIdx = firstFrame + row;
450 AppendFrameRow(sb, frameIdx, written, () =>
451 {
452 var values = new long?[N];
453 for (int i = 0; i < N; i++)
454 {
455 if (columns[i] == null) { values[i] = null; continue; }
456 values[i] = (long)Math.Round(columns[i][row]);
457 }
458 return values;
459 });
460 written++;
461 }
462 EndSidecar(sb, written);
463 File.WriteAllText(sidecarPath, sb.ToString());
464 }

466 // --- sidecar JSON helpers ---
467 private static void BeginSidecar(StringBuilder sb, int frameIndexOffset)
468 {
469 sb.Append("{\n");
470 sb.AppendFormat(" \"schemaVersion\": {0},\n", CountersSchemaVersion);
471 sb.AppendFormat(" \"frameIndexOffset\": {0},\n", frameIndexOffset);
472 sb.Append(" \"counters\": [");
473 for (int i = 0; i < SchemaFieldNames.Length; i++) { if (i > 0) sb.Append(','); sb.Append('"').Append(SchemaFieldNames[i]).Append('"'); }
474 sb.Append("],\n \"frames\": [\n");
475 }
476 private static void AppendFrameRow(StringBuilder sb, int frameIdx, int writtenSoFar, Func<long?[]> compute)
477 {
478 if (writtenSoFar > 0) sb.Append(",\n");
479 sb.Append(" {\"frameIndex\": ").Append(frameIdx).Append(", \"values\": [");
480 var values = compute();
481 for (int i = 0; i < values.Length; i++) { if (i > 0) sb.Append(','); if (values[i].HasValue) sb.Append(values[i].Value); else sb.Append("null"); }
482 sb.Append("]}");
483 }
484 private static void EndSidecar(StringBuilder sb, int writtenFrames) { sb.Append("\n ],\n"); sb.AppendFormat(" \"frameCount\": {0}\n}}\n", writtenFrames); }
485 private static string EmptySidecarJson(int frameIndexOffset, int writtenFrames) { var sb = new StringBuilder(); BeginSidecar(sb, frameIndexOffset); EndSidecar(sb, writtenFrames); return sb.ToString(); }
