708 private void RecvMessages(bool tryCopyToThreadBuffer = false)
709 {
710 if (pauseRead_in_cacheMode)
711 {
712 return;
713 }
714
715 var begin = Time.realtimeSinceStartup;
716 var length = 0;
717 var ret = ReadData(ref m_recvBuffer, ref length);
718
719 while (ret)
720 {
721 unsafe
722 {
723 if (length <= 0 || length > m_cacheBuffer.Length)
724 {
725 Debug.LogError($"[TServer] one package length invalide: len = {length}, buffer.len = {m_cacheBuffer.Length}");
726 }
727 else
728 {
729 if (m_cacheStartIndex > m_cacheEndIndex && m_cacheEndIndex + length >= m_cacheStartIndex)
730 {
731 #if ENABLE_PROFILER
732 m_samplerDecodeMessages.Begin();
733 #endif
734 Debug.LogWarning($"[TServer] cache buffer DecodeMessages(true)");
735
736 DecodeMessages(true);
737 #if ENABLE_PROFILER
738 m_samplerDecodeMessages.End();
739 #endif
740 continue;
741 }
742
743 if (m_cacheEndIndex + length < m_cacheBuffer.Length)
744 {
745 bool toThreadBuffer = false;
746 if (tryCopyToThreadBuffer)
747 {
748 toThreadBuffer = TryCopyToThreadBuffer(m_recvBuffer, length);
749 }
750
751 if (!toThreadBuffer)
752 {
753 // 尾部空间足够
754 fixed (void* pFrom = m_recvBuffer, pTo = &m_cacheBuffer[m_cacheEndIndex])
755 {
756 System.Buffer.MemoryCopy(pFrom, pTo, m_cacheBuffer.Length - m_cacheEndIndex, length);
757 m_cacheLengthQueue.Enqueue(new CachePackageArg()
758 {
759 len = length,
760 time = Time.realtimeSinceStartup
761 });
762 m_cacheEndIndex += length;
763 }
764 }
765 }
766 else if (m_cacheEndIndex + length == m_cacheBuffer.Length)
767 {
768 bool toThreadBuffer = false;
769 if (tryCopyToThreadBuffer)
770 {
771 TryCopyToThreadBuffer(m_recvBuffer, length);
772 }
773
774 if (!toThreadBuffer)
775 {
776 // 尾部空间刚好
777 fixed (void* pFrom = m_recvBuffer, pTo = &m_cacheBuffer[m_cacheEndIndex])
778 {
779 System.Buffer.MemoryCopy(pFrom, pTo, length, length);
780 m_cacheLengthQueue.Enqueue(new CachePackageArg()
781 {
782 len = length,
783 time = Time.realtimeSinceStartup
784 });
785 m_cacheEndIndex = 0;
786 }
787 }
788 }
789 else
790 {
791 bool toThreadBuffer = false;
792 if (tryCopyToThreadBuffer)
793 {
794 TryCopyToThreadBuffer(m_recvBuffer, length);
795 }
796
797 if (!toThreadBuffer)
798 {
799 Debug.LogWarning($"[TServer] cache buffer return to head");
800 // 【尾部空间不足】 且头部空间足够，和DecodeMessage读RingBuffer的操作逻辑保持一致。
801 if (m_cacheStartIndex > length)
802 {
