95function cls:OnUpdate()
96    if self.curCountDownType then
97        local func = RefreshTimeFunc[self.curCountDownType]
98        if not func or not func(self) then
99            self.curCountDownType = nil
100            -- 当前倒计时结束不等于所有倒计时结束，重新选择仍有效的类型。
101            self:Refresh()
102        end
103    end
104
105    self:SetUpdateEnable(self.curCountDownType ~= nil)
106end

114function cls:OnRefresh()
115    local mapEntityData = self.mapEntity and self.mapEntity.mapEntityData
116    if mapEntityData and mapEntityData.CanShowTimeText and not mapEntityData:CanShowTimeText() then
117        self.curCountDownType = nil
118        self:SetUpdateEnable(false)
119        return
120    end
121    -- entity 自定义钩子（仿 BattleHeadItemNameText.InitName 中的 CanShowName）：返回 false 时整段倒计时容器隐藏。
122    -- 用于华夏统帅建造资源分城隐藏期等需要把名字、头像、倒计时一起隐掉的场景。
123    if self.mapEntity and self.mapEntity.CanShowTimeText and not self.mapEntity:CanShowTimeText() then
124        self.curCountDownType = nil
125        self:SetUpdateEnable(false)
126        return
127    end
128    --- Reset animation state on refresh to avoid stale animation from previous usage
129    if self.isPlayingOutAnim then
130        if self.animation then
131            self.animation:Stop()
132        end
133        self.isPlayingOutAnim = false
134    end
135    --- 取消之前的退出动画异步协程，防止延迟隐藏覆盖新的显示状态
136    AsyncStopAllBinded(self)
137    self.curCountDownType = nil
138    local ctrlData = self:GetDynamicCtrlData()
139    ctrlData.m_LabType:SetVisible(false)
140    for type, func in pairs(RefreshTimeFunc) do
141        
142        ProfilerUtil.SamplerBegin(samplerDic[type])
143        local ret = func(self)
144        ProfilerUtil.SamplerEnd(samplerDic[type])
145        if ret then
146            self.curCountDownType = type
147            if self.useMUI then
148                if RefreshTimeShowLabType[type] == true then
149                    ctrlData.m_LabType:SetVisible(RefreshTimeShowLabType[type] == true)
150                end
151            else
152                if ctrlData and ctrlData.m_LabType then
153                    ctrlData.m_LabType:SetVisible(RefreshTimeShowLabType[type] == true)
154                end
155            end
156
157            break
158        end
159    end
160
161    self:SetUpdateEnable(self.curCountDownType ~= nil)
162end

182function cls:RefreshStormTime()
183    if self.mapEntity.entityType ~= CfgNs.WorldEntityType.WET_Tornado then
184        return false
185    end
186    if not self.mapEntity.mapEntityData.stormAttr or self.mapEntity.mapEntityData.stormAttr.status ~= CfgNs.TornadoLifecycleStatus.TLS_Creation then
187        return false
188    end
189    local curTime = mgr.time.serverUTCTsInMs
190    local endTime = self.mapEntity.mapEntityData.stormAttr.statusEndTime
191    if endTime <= curTime then return false end
192    local time = endTime - curTime
193    local ctrlData = self:GetDynamicCtrlData()
194    --ctrlData.m_ProgTime:SetVisible(false)
195    --ctrlData.m_ProgTimeBg?:SetVisible(false)
196    time = time / 1000
197    time = math.floor(time)
198    local date = os.date("!*t", time)
199    
200    if date.min <= 0 then
201        ctrlData.m_LabTime:SetTextContent(mgr.text:GetText("Text_TornadoTime1", tostring(date.sec)))
202    else
203        ctrlData.m_LabTime:SetTextContent(mgr.text:GetText("Text_TornadoTime2", tostring(date.min), tostring(date.sec)))
204    end
205    return true
206end

208function cls:RefreshChantTime()
209    local skillInfo = self.mapEntity.mapEntityData.skillInfo
210    if not skillInfo then return false end
211
212    local curTime = mgr.time:TransTickToTimeInMs(mgr.time.gameTick)
213    local endTime = mgr.time:TransTickToTimeInMs(skillInfo.endTick or 0)
214    if endTime <= curTime then return false end
215
216    local time = endTime - curTime
217    local progress = 1 - math.min(1, time / skillInfo.durationMillis)
218    local ctrlData = self:GetDynamicCtrlData()
219    self:RefreshLightEff(ctrlData, progress)
220    ctrlData.m_LabTime:SetTextContent(TimeUtil.FormatCountdown(time / 1000))
221    
222    return true
223end

225function cls:RefreshFreshTime()
226    if self.mapEntity.mapEntityData == nil or self.mapEntity.mapEntityData.moveAttr == nil then return false end
227    local moveAttr = self.mapEntity.mapEntityData.moveAttr
228    local endTime = mgr.time:TransTickToTimeInMs(moveAttr.endTick)
229    local leftTime = mgr.time.scale:GetSceneLeftTime(endTime)
230    local seconds = math.floor(leftTime / 1000)
231    if leftTime <= 0 or endTime <= 0 or seconds <= 0 then
232        return false
233    end
234    
235    local ctrlData = self:GetDynamicCtrlData()
236    
237    ProfilerUtil.SamplerBegin(samplerDic[6])
238    local _time = TimeUtil.FormatCountdown(seconds)
239    ProfilerUtil.SamplerEnd(samplerDic[6])
240    
241    
242    ProfilerUtil.SamplerBegin(samplerDic[7])
243    ctrlData.m_LabTime:SetTextContent(_time)
244    ProfilerUtil.SamplerEnd(samplerDic[7])
245    
246    return true
247end

251function cls:RefreshOccupyTime()
252    local ctrlData = self:GetDynamicCtrlData()
253    local data = self.mapEntity.mapEntityData.armyAttr?.operateResourceAttr or GlobalEmptyTable
254    if data.opType ~= CfgNs.ArmyOperateResourceType.AORT_Occupying then
255        return false
256    end
257
258    -- 优先用地块 occupyProgressAttr；地块 FSM 缺失/被 geo deleted 清掉时 leftTime 会为 0，需兜底到部队 op 时间
259    if (data.resourceId or 0) > 0 then
260        local resourceEntityData = mgr.mapCore.DataCtrl:GetEntityDataByGridIndexAndType(data.resourceId, CfgNs.WorldEntityType.WET_Resource)
261        if resourceEntityData then
262            local value, leftTime = resourceEntityData:GetOccupyProgressValue(self.mapEntity.mapEntityData.ownerUid)
263            if leftTime > 0 then
264                self:RefreshLightEff(ctrlData, value)
265                ctrlData.m_LabTime:SetTextContent(TimeUtil.FormatCountdown(leftTime / 1000))
266                return true
267            end
268        end
269    end
270
271    local time = data.opEndTimeMs - serverUTCTsInMs
272    if time <= 0 then
273        self:RefreshLightEff(ctrlData, 0)
274        ctrlData.m_LabTime:SetTextContent(" ")
275        return false
276    end
277    local duration = (data.opEndTimeMs - data.opStartTimeMs)
278    local progress = duration > 0 and (time / duration) or 0
279    self:RefreshLightEff(ctrlData, progress)
280    ctrlData.m_LabTime:SetTextContent(TimeUtil.FormatCountdown(time / 1000))
281
282    return true
283end

318function cls:RefreshFightingWall()
319    local data = self.mapEntity and self.mapEntity.mapEntityData
320    if not data or not data.GetArmyStateExpireTimeMs then return false end
321    -- 与部队初始化一致：有效 fightState 优先，否则回退 armyState；旧宿主兼容 GetFightState。
322    local getState = data.GetArmyState or data.GetFightState
323    if not getState or getState(data) ~= CfgNs.ArmyState.MS_FightingWall then
324        return false
325    end
326
327    local endTime = data:GetArmyStateExpireTimeMs()
328    if not endTime then
329        return false
330    end
331
332    local curTime = mgr.time.serverUTCTsInMs
333    local remainTime = endTime - curTime
334    if remainTime <= 0 then
335        -- mgr.log:Error("[Chaiqian] entityId : %s, remainTime: %s, endTime: %s, curTime: %s,", self.mapEntity.entityId, remainTime, endTime, curTime)
336        return false
337    end
338
339    local timeStr = TimeUtil.FormatCountdownShort(remainTime / 1000)
340    ---@type Pan_BattleHead_Time_UGUI_CtrlData
341    local ctrlData = self:GetDynamicCtrlData()
342    ctrlData.m_LabTime:SetTextContent(timeStr)
343    
344    return true
345end
