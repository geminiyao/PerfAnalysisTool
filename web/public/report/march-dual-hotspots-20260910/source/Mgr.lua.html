233---@generic T
234---@param table T
235---@return T
236function Query (table)
237    local value = table
238    local ret = {}
239    ret.Get = function (v1, v2)
240        local dv = v1
241        if ret == v1 then
242            dv = v2
243        end
244        local tv = value
245        value = table
246        return tv ~= nil and tv or dv
247    end
248    setmetatable(ret, {
249        __index = function (_, k)
250            local b = type(value) == "table"
251            value = b and value[k] or nil
252            return ret
253        end
254    })
255    return ret
256end

296function cls:BeginAddLoggedInScopedMgrs()
297  if self._loggedInScopeMgrFlagDic and ((next(self._loggedInScopeMgrFlagDic) ~= nil)) then
298    local names = ""
299    for child,_ in pairs(self._loggedInScopeMgrFlagDic) do
300      names = names .. child.name .. "\n"
301    end
302    self.log:Error("_loggedInScopeMgrFlagDic must be empty when calling BeginAddLogicMgr()! keys:\n%s",
303      names)
304    return
305  end
306  self._addingLoggedInScopedMgr = true
307  --- # create a LoggedIn Scoped Mgr's table, set to Key Weak
308  self._loggedInScopeMgrFlagDic = {}
309  setmetatable(self._loggedInScopeMgrFlagDic, {__mode="k"})
310end

346function cls:removeLoggedInMgrs()
347  if self._loggedInScopeMgrFlagDic and next(self._loggedInScopeMgrFlagDic)~=nil then
348    local childrenClone = self:GetChildren()
349    for i = #childrenClone,1,-1 do
350      local child = childrenClone[i]
351      if self._loggedInScopeMgrFlagDic[child] and (not child.__removing) then
352        -- # Mark as removing
353        -- self.log:Error("remove child by scope named: " .. tostring(child.name))
354        child.__removing = true
355        self._loggedInScopeMgrFlagDic[child] = nil
356        -- # Call Logout if 'loggedIn'
357        if child.loggedIn then
358          local succ,err = xpcall(child.Logout, debug.traceback, child, true)
359          if not succ then
360            self.log:Error(err)
361          end
362        end
363        -- # Call CleanUp if 'inited'
364        if child.inited then
365          local succ,err = xpcall(child.CleanUp, debug.traceback, child, true)
366          if not succ then
367            self.log:Error(err)
368          end
369        end
370        -- # Remove name access
371        --- TODO thy extract duplicated logic (with Entity.lua)
372        if not StrUtil.IsNullOrEmpty(child.name) and self[child.name]==child then
373          self[child.name] = nil
374        end
375        -- self:TriggerFrameEnd()
376      end
377    end
378    self:SetRemoveDirty()
379  end
380end
