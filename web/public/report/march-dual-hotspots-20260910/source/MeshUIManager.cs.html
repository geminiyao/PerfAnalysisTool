186        {
187            var textureRender = new MUIDefaultRenderer();
188            textureRender.renderBatchName = "Debug_MUI_Texture";
189            textureRender.Init(1000, 10000, m_root);
190            textureRender.SetShader(resource.shaderMeshUITexture);
191            textureRender.SetAtlas(resource.atlasBattleBubble);
192            textureRender.SetRenderingLayerMask(1 << (int) RenderLayerMaskEnum.World);
193            textureRender.SetRenderQueue(resource.renderQueueDefault);
194            if (!renderers.RegisteRenderer(textureRender))
