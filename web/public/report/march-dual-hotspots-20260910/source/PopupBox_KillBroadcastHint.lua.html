171function cls:_OnTick()

183    --1.处理显示中的过期项：过期则释放，并立即播放该部队队列里的下一条(复用槽位)
184    local curTime = mgr.time.realT
185    local expired = {}
186    for entityId, startTime in pairs(self.KillBroadcastUIStartTime) do
187        if startTime ~= nil and curTime - startTime >= self.showDuration then
188            table.insert(expired, entityId)
189        end
190    end
191    for _, entityId in ipairs(expired) do
192        self:ReleaseKillBroadcastByEntityId(entityId)
193        self:_tryShowNextForArmy(entityId)
194    end
195
196    --2.为有排队事件但未在显示的部队填充空闲槽位
197    self:_fillDisplaySlots()
198
199    --3.无显示且队列全空 -> 停 tick
200    if TblUtil.IsEmpty(self.KillBroadcastUI) and self:_isAllQueueEmpty() then
201        self:StopTickHandler()
202    end
203end

249--按到达时间过滤同一部队的 godlike：1s 滑动窗口内最多放行 GODLIKE_MAX_PER_WINDOW 个
250--返回 true 表示放行(可入队)，false 表示丢弃
251function cls:_acceptGodlikeOnArrival(entityId)
252    local now = mgr.time.realT
253    local last = self.godlikeLastArrivalTime[entityId]
254    --距上一次 godlike 到达超过窗口 -> 清零重新计算
255    if last == nil or (now - last) > self.GODLIKE_RESET_INTERVAL then
256        self.godlikeArrivalCount[entityId] = 0
257    end
258    --只要有 godlike 到达就刷新窗口(持续刷屏则持续丢弃)
259    self.godlikeLastArrivalTime[entityId] = now
260
261    local count = self.godlikeArrivalCount[entityId] or 0
262    if count < self.GODLIKE_MAX_PER_WINDOW then
263        self.godlikeArrivalCount[entityId] = count + 1
264        return true
265    end
266    return false
267end

282    if not queue then
